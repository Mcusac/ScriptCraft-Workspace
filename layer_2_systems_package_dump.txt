package com.biocraft.alien.layer_2_systems.level_0.ai;

import com.biocraft.alien.layer_0_domain.level_0.spatial.BlockPosRef;
import java.util.function.IntUnaryOperator;
import java.util.function.Predicate;

/**
 * Expanding-radius block position search (algorithm only; probes supplied by L3).
 */
public final class BlockPosSearchPolicy {

    private BlockPosSearchPolicy() {}

    public static BlockPosRef searchInRadius(
            BlockPosRef center,
            double maxRadius,
            int attemptsPerRadius,
            IntUnaryOperator randomNextInt,
            Predicate<BlockPosRef> predicate) {
        if (center == null || randomNextInt == null || predicate == null) {
            return null;
        }

        for (int radius = 5; radius <= maxRadius; radius += 5) {
            for (int i = 0; i < attemptsPerRadius; i++) {
                BlockPosRef candidate = center.offset(
                        randomNextInt.applyAsInt(radius * 2) - radius,
                        randomNextInt.applyAsInt(4) - 2,
                        randomNextInt.applyAsInt(radius * 2) - radius);
                if (predicate.test(candidate)) {
                    return candidate;
                }
            }
        }
        return null;
    }
}

import com.biocraft.alien.layer_1_infrastructure.level_1.entity.EntityConfigStorage;
import com.biocraft.alien.layer_1_infrastructure.level_0.registry.ConfigContractException;

/**
 * Resolves AI goal slot priorities from entity JSON via {@link EntityConfigStorage}.
 * No Java fallbacks — missing keys throw {@link ConfigContractException}.
 *
 * <p>TODO(GAMEPLAY-TARGETING-001): Externalize combat target-priority ownership only when retuning combat.
 */
public final class GoalPriorityResolver {

    private static final String RESOLVER = "GoalPriorityResolver";

    private GoalPriorityResolver() {
        throw new UnsupportedOperationException("GoalPriorityResolver is a utility class and cannot be instantiated");
    }

    /**
     * Returns the configured priority for a goal key on the given xenomorph type.
     *
     * @param type    species owning the priority map
     * @param goalKey key in {@code ai_goals.goal_priorities}
     * @return priority integer (lower = higher precedence in Minecraft goal selector)
     */
    public static int priority(XenomorphType type, String goalKey) {
        EntityConfig config = EntityConfigStorage.getConfig(type);
        if (config.aiGoals == null) {
            throw new ConfigContractException(RESOLVER, type.name(), "ai_goals", "required");
        }
        if (config.aiGoals.goalPriorities == null) {
            throw new ConfigContractException(RESOLVER, type.name(), "ai_goals.goal_priorities", "required");
        }
        Integer priority = config.aiGoals.goalPriorities.get(goalKey);
        if (priority == null) {
            throw new ConfigContractException(
                    RESOLVER, type.name(), "ai_goals.goal_priorities." + goalKey, "required");
        }
        return priority;
    }
}

package com.biocraft.alien.layer_2_systems.level_0.ai;

/**
 * Pure suffocation decision policy for xenomorph survival AI.
 * Thresholds are ratios of current air supply to max air supply (0.0–1.0).
 */
public final class SurvivalSuffocationPolicy {

    private SurvivalSuffocationPolicy() {
        throw new UnsupportedOperationException("SurvivalSuffocationPolicy is a utility class");
    }

    public static boolean isSuffocating(
            boolean inWall,
            boolean stuck,
            int airSupply,
            int maxAirSupply,
            double lowThreshold,
            double criticalThreshold) {
        if (inWall) {
            return true;
        }
        if (maxAirSupply <= 0) {
            return false;
        }

        double airRatio = (double) airSupply / maxAirSupply;
        if (airRatio <= criticalThreshold) {
            return true;
        }
        return stuck && airRatio <= lowThreshold;
    }
}

package com.biocraft.alien.layer_2_systems.level_0.ai;

import com.biocraft.alien.layer_0_domain.level_0.spatial.Vec3Ref;
import java.util.function.IntUnaryOperator;
import java.util.function.Predicate;

/**
 * Directional Vec3 position search (algorithm only; validity probes supplied by L3).
 */
public final class Vec3SearchPolicy {

    private Vec3SearchPolicy() {}

    public static Vec3Ref search(
            Vec3Ref origin,
            Vec3Ref baseDirection,
            double baseDistance,
            int maxAttempts,
            IntUnaryOperator randomNextInt,
            Predicate<Vec3Ref> predicate) {
        if (origin == null || baseDirection == null || randomNextInt == null || predicate == null) {
            return null;
        }

        Vec3Ref direction = baseDirection.normalize();
        for (int i = 0; i < maxAttempts; i++) {
            Vec3Ref candidate = origin.add(direction.scale(baseDistance + randomNextInt.applyAsInt(8)));
            if (predicate.test(candidate)) {
                return candidate;
            }
            direction = direction.yRot((float) (Math.PI / 4));
        }
        return null;
    }
}

/**
 * Helper for layer_2_systems to query block rules from infrastructure config.
 *
 * FILE STATUS: NEW
 *
 * @author BioCraft
 * @since 0.1.0
 */
public final class BlockRuleHelper {

    private static final String HELPER = "BlockRuleHelper";

    private BlockRuleHelper() {
        throw new UnsupportedOperationException(
                "BlockRuleHelper is a utility class and cannot be instantiated");
    }

    /**
     * Check if a block is unbreakable per entity_block_interaction.json.
     *
     * @param blockId registry ID such as "minecraft:bedrock"
     * @return true if this block cannot be broken
     */
    public static boolean isUnbreakable(String blockId) {
        if (blockId == null || blockId.isBlank()) {
            return false;
        }

        if (!EntityBlockInteractionConfigStorage.isLoaded()) {
            ModLogger.warn(HELPER, "Block rules not loaded; treating block as breakable");
            return false;
        }

        return EntityBlockInteractionConfigStorage.getConfig().isUnbreakable(blockId);
    }

    /**
     * Check if a block is explicitly whitelisted as breakable.
     *
     * @param blockId registry ID
     * @return true if explicitly breakable in config
     */
    public static boolean isExplicitlyBreakable(String blockId) {
        if (blockId == null || blockId.isBlank()) {
            return false;
        }

        if (!EntityBlockInteractionConfigStorage.isLoaded()) {
            return false;
        }

        return EntityBlockInteractionConfigStorage.getConfig().isExplicitlyBreakable(blockId);
    }
}

package com.biocraft.alien.layer_2_systems.level_0.combat;

import com.biocraft.alien.layer_0_domain.level_1.blocks.BlockBreakPolicy;
import com.biocraft.alien.layer_0_domain.level_0.blocks.BlockBreakRules;
import com.biocraft.alien.layer_0_domain.level_0.spatial.BlockPosRef;
import com.biocraft.alien.layer_0_domain.level_0.spatial.WorldRef;
import com.biocraft.alien.layer_0_domain.level_1.world.WorldMutationPort;
import com.biocraft.alien.layer_1_infrastructure.level_0.block.EntityBlockInteractionConfig;
import com.biocraft.alien.layer_1_infrastructure.level_1.block.EntityBlockInteractionConfigStorage;
import com.biocraft.alien.layer_1_infrastructure.level_0.platform.ModLogger;
import com.biocraft.alien.layer_1_infrastructure.level_1.block.EntityBlockInteractionRulesMapper;

/**
 * Block damage orchestration for xenomorphs breaking blocks.
 *
 * Instant break when {@link BlockBreakPolicy} allows. Damage settings in
 * entity_block_interaction.json are validated but not yet applied as multi-hit HP.
 *
 * FILE STATUS: STABLE
 *
 * @author BioCraft
 * @since 0.1.0
 */
public class BlockDamageOrchestrator {

    private static final String ORCHESTRATOR = "BlockDamageOrchestrator";
    private static boolean isEnabled = false;

    private BlockDamageOrchestrator() {
    }

    /**
     * Initialize block damage system from loaded entity-block interaction config.
     */
    public static void initialize() {
        try {
            if (!EntityBlockInteractionConfigStorage.isLoaded()) {
                ModLogger.warn(ORCHESTRATOR, "Entity-block interaction config not loaded yet");
                isEnabled = false;
                return;
            }

            EntityBlockInteractionConfig config = EntityBlockInteractionConfigStorage.getConfig();
            isEnabled = config.enabled != null && config.enabled;

            if (isEnabled) {
                ModLogger.info(ORCHESTRATOR, "Block Damage System enabled");
            } else {
                ModLogger.info(ORCHESTRATOR, "Block Damage System disabled");
            }
        } catch (Exception e) {
            ModLogger.warn(ORCHESTRATOR, "Failed to initialize Block Damage System: " + e.getMessage());
            isEnabled = false;
        }
    }

    public static boolean isEnabled() {
        return isEnabled;
    }

    /**
     * Attempt to break a block when policy allows.
     *
     * @param mutation      domain mutation port for world operations
     * @param world         domain world reference
     * @param pos           block position
     * @param blockId       registry id such as {@code minecraft:glass}
     * @param entityTypeKey config key such as {@code drone} or {@code queen}
     * @param actorLabel    logging label for the breaking entity
     * @return true when the block was broken
     */
    public static boolean attemptBlockDamage(
            WorldMutationPort mutation,
            WorldRef world,
            BlockPosRef pos,
            String blockId,
            String entityTypeKey,
            String actorLabel) {

        if (!isEnabled || mutation == null || world == null || pos == null
                || blockId == null || blockId.isBlank()) {
            return false;
        }

        if (!EntityBlockInteractionConfigStorage.isLoaded()) {
            ModLogger.warn(ORCHESTRATOR, "Config not loaded; cannot break blocks");
            return false;
        }

        EntityBlockInteractionConfig config = EntityBlockInteractionConfigStorage.getConfig();
        if (config == null) {
            return false;
        }

        boolean canBreak = config.canEntityBreakBlocks(entityTypeKey);
        boolean unrestricted = config.hasUnrestrictedBreakScope(entityTypeKey);
        BlockBreakRules rules = EntityBlockInteractionRulesMapper.toDomainRules(config);

        if (!BlockBreakPolicy.mayBreakBlock(blockId, canBreak, unrestricted, rules)) {
            return false;
        }

        boolean broken = mutation.breakBlock(world, pos);
        if (broken) {
            ModLogger.debug(ORCHESTRATOR,
                    actorLabel + " broke block " + blockId + " at " + pos);
        }
        return broken;
    }

    /**
     * Reload configuration and re-initialize.
     */
    public static void reload() {
        initialize();
    }
}

package com.biocraft.alien.layer_2_systems.level_0.detection;

import java.util.Objects;

import com.biocraft.alien.layer_0_domain.level_0.observation.MovementState;

/**
 * Tracker consumer policy: relevance and ignore filters over observation facts.
 *
 * <p>Does not produce facts and does not gate on organism type.
 */
public final class MotionTrackerConsumerPolicy {

    private MotionTrackerConsumerPolicy() {
    }

    /**
     * Presentation-agnostic traits derived by the Minecraft adapter.
     */
    public record EntityTraits(
            boolean player,
            boolean friendlyMob,
            boolean armorStand,
            boolean spectator,
            MovementState movement
    ) {
        public EntityTraits {
            movement = Objects.requireNonNull(movement, "movement required");
        }
    }

    /**
     * Consumer ignore / relevance settings (from tracker config).
     */
    public record IgnoreSettings(
            boolean ignorePlayers,
            boolean ignoreFriendlyMobs,
            boolean ignoreArmorStands,
            boolean ignoreSpectators,
            boolean reportStill
    ) {
    }

    public static boolean shouldReport(EntityTraits traits, IgnoreSettings settings) {
        Objects.requireNonNull(traits, "traits");
        Objects.requireNonNull(settings, "settings");
        if (settings.ignoreSpectators() && traits.spectator()) {
            return false;
        }
        if (settings.ignorePlayers() && traits.player()) {
            return false;
        }
        if (settings.ignoreArmorStands() && traits.armorStand()) {
            return false;
        }
        if (settings.ignoreFriendlyMobs() && traits.friendlyMob()) {
            return false;
        }
        if (!settings.reportStill() && traits.movement() == MovementState.STILL) {
            return false;
        }
        return true;
    }

    /**
     * @param entitiesReported count of entities shown to the player this scan
     * @param alertToneEntityThreshold from config — alert when count is strictly greater
     */
    public static boolean shouldPlayAlertTone(int entitiesReported, int alertToneEntityThreshold) {
        return entitiesReported > alertToneEntityThreshold;
    }
}

package com.biocraft.alien.layer_2_systems.level_0.growth;

import com.biocraft.alien.layer_0_domain.level_0.entity.EntityThreatLevel;
import com.biocraft.alien.layer_0_domain.level_1.entity.XenomorphType;
import com.biocraft.alien.layer_1_infrastructure.level_3.lifecycle.LifecycleConfig;

/**
 * Resolver for xenomorph entity form transitions based on lifecycle stage.
 * Maps entity lifecycle data to appropriate xenomorph form (drone, queen, etc.).
 *
 * <p><strong>Authority:</strong> {@code lifecycle.json} {@code form_thresholds} via {@link LifecycleConfig}.
 *
 * <p><strong>Runtime note:</strong> Live entity form identity uses {@code XenomorphFormQueryPort}
 * (NeoForge adapter + entity type). This resolver is pre-wired infrastructure for future
 * maturation/threat-driven form promotion — no production caller yet (MIG-DEF-005 closed).
 *
 * @author BioCraft
 * @since 0.1.0
 */
public class EntityFormResolver {
    
    /**
     * Resolve xenomorph form based on lifecycle parameters.
     * Determines which form (drone, queen, praetor, etc.) the entity should take.
     * 
     * @param baseType Base xenomorph type from host
     * @param maturationStage Current maturation stage (0-100)
     * @param threatLevel Current threat level assessment
     * @return Resolved xenomorph form
     */
    public XenomorphType resolveForm(XenomorphType baseType, int maturationStage, EntityThreatLevel threatLevel) {
        if (baseType == null) {
            return XenomorphType.DRONE;
        }
        
        if (maturationStage < LifecycleConfig.getEarlyStageExclusiveThreshold()) {
            return XenomorphType.DRONE;
        }
        
        if (maturationStage < LifecycleConfig.getMidStageExclusiveThreshold()) {
            return XenomorphType.DRONE;
        }
        
        if (maturationStage < LifecycleConfig.getLateStageExclusiveThreshold()) {
            if (threatLevel == EntityThreatLevel.HIGH || threatLevel == EntityThreatLevel.CRITICAL) {
                return XenomorphType.PRAETOR;
            }
            return XenomorphType.DRONE;
        }
        
        if (maturationStage >= LifecycleConfig.getQueenMaturationMinThreshold()
                && threatLevel == EntityThreatLevel.CRITICAL) {
            return XenomorphType.QUEEN;
        }
        
        return XenomorphType.PRAETOR;
    }
    
    /**
     * Check if form transition is allowed at current stage.
     * 
     * @param currentForm Current form
     * @param targetForm Target form to transition to
     * @param maturationStage Current maturation (0-100)
     * @return true if transition is allowed
     */
    public boolean canTransition(XenomorphType currentForm, XenomorphType targetForm, int maturationStage) {
        if (currentForm == targetForm) {
            return true;
        }
        
        if (currentForm == XenomorphType.DRONE
                && targetForm == XenomorphType.PRAETOR
                && maturationStage >= LifecycleConfig.getDroneToPraetorMinThreshold()) {
            return true;
        }
        
        if (currentForm == XenomorphType.PRAETOR

import com.biocraft.alien.layer_0_domain.level_0.nest.EggLayingPolicy;

import com.biocraft.alien.layer_0_domain.level_2.xenomorph.XenomorphFormQueryPort;

import java.util.List;
import java.util.function.Predicate;

/**
 * Orchestration workflow for egg laying behavior.
 * Encapsulates timing, location search, and policy enforcement.
 *
 * Static utility class — no instance state.
 * Each AI behavior maintains its own timer and calls these stateless methods.
 *
 * Boundaries:
 * - Does NOT import Minecraft types
 * - Does NOT perform block placement
 * - Does NOT own timing state
 */
public final class EggLayingWorkflow {

    private EggLayingWorkflow() {
    }

    /**
     * Advance a timer by one tick.
     *
     * @param currentTimer current timer value
     * @param interval threshold for reset
     * @return TickResult with next timer value and whether threshold was reached
     */
    public static TickResult advanceTimer(int currentTimer, int interval) {
        int next = currentTimer + 1;
        return new TickResult(next >= interval ? 0 : next, next >= interval);
    }

    /**
     * Find a suitable candidate location for egg placement within limits.
     * Searches in expanding rings from center, testing with provided predicate.
     *
     * @param center search center position
     * @param limits search area and density limits
     * @param suitability test for location viability
     * @return candidate position or null if none found
     */
    public static BlockPosRef findCandidateLocation(BlockPosRef center, Limits limits, Predicate<BlockPosRef> suitability) {
        if (center == null || limits == null || suitability == null) {
            return null;
        }
        int radius = limits.eggPlacementRadius();
        for (int ring = 0; ring <= radius; ++ring) {
            for (int x = -ring; x <= ring; ++x) {
                for (int z = -ring; z <= ring; ++z) {
                    if (Math.abs(x) != ring && Math.abs(z) != ring) {
                        continue;
                    }
                    for (int y = -limits.verticalSearchRange(); y <= limits.verticalSearchRange(); ++y) {
                        BlockPosRef candidate = new BlockPosRef(center.x() + x, center.y() + y, center.z() + z);
                        if (!suitability.test(candidate)) {
                            continue;
                        }
                        return candidate;
                    }
                }
            }
        }
        return null;
    }

    /**
     * Count ovomorph eggs present in the search area via world entity scan.
     */
    public static int countEggsInArea(
            WorldQueryPort query,
            XenomorphFormQueryPort forms,
            WorldRef world,
            BlockPosRef center,
            Limits limits) {
        if (query == null || forms == null || world == null || center == null || limits == null) {
            return 0;
        }
        List<EntityScanHit> hits = query.scanLivingEntities(world, regionAround(center, limits.searchRadius(), limits.verticalSearchRange()));
        int count = 0;
        for (EntityScanHit hit : hits) {
            if (isOvomorph(forms, world, hit)) {
                count++;
            }
        }
        return count;
    }

    /**
     * Check if egg laying is allowed based on density limits.
     *
     * @param eggCount count in area
     * @param chunkEggCount count in chunk
     * @param limits policy limits
     * @return true if laying is permitted
     */
    public static boolean canLayInArea(int eggCount, int chunkEggCount, Limits limits) {
        if (limits == null) {
            return false;
        }
        return EggLayingPolicy.canLayEgg(eggCount, limits.maxEggsInArea())
                && EggLayingPolicy.isChunkBelowEggLimit(chunkEggCount, limits.maxEggsPerChunk());
    }

    /**
     * Check if location is clear of nearby ovomorph eggs.
     */
    public static boolean isLocationClearOfEggs(
            WorldQueryPort query,
            XenomorphFormQueryPort forms,
            WorldRef world,
            BlockPosRef pos,
            int minDistance) {
        if (pos == null) {
            return false;
        }
        if (query == null || forms == null || world == null || minDistance <= 0) {
            return true;
        }
        double maxDistSq = (double) minDistance * (double) minDistance;
        List<EntityScanHit> hits = query.scanLivingEntities(world, regionAround(pos, minDistance, 2));
        for (EntityScanHit hit : hits) {
            if (hit.distanceSq() < maxDistSq && isOvomorph(forms, world, hit)) {
                return false;
            }
        }
        return true;
    }

    private static boolean isOvomorph(XenomorphFormQueryPort forms, WorldRef world, EntityScanHit hit) {
        if (hit == null || hit.entityId() == null) {
            return false;
        }
        return forms.resolveForm(new LivingEntityRef(hit.entityId(), world)) == XenomorphType.OVOMORPH;
    }

    private static AxisAlignedRegion regionAround(BlockPosRef center, int horizontalRadius, int verticalRange) {
        return new AxisAlignedRegion(
                new BlockPosRef(center.x() - horizontalRadius, center.y() - verticalRange, center.z() - horizontalRadius),
                new BlockPosRef(
                        center.x() + horizontalRadius + 1,
                        center.y() + verticalRange + 1,
                        center.z() + horizontalRadius + 1));
    }

    /**
     * Result of a timer tick operation.
     */
    public record TickResult(int nextTimer, boolean shouldAttemptLay) {
    }

    /**
     * Search and density limits for egg laying.
     */
    public record Limits(
            int searchRadius,
            int maxEggsPerChunk,
            int maxEggsInArea,
            int eggPlacementRadius,
            int minDistanceBetweenEggs,
            int verticalSearchRange
    ) {
    }
}

    }

    public record BlockOffset(int x, int y, int z) {
    }

    /**
     * Plan block placements for one nest/hive phase.
     *
     * <p>TODO(FEATURE-NEST-002): prefer open-air footprints; reduce dense resin carpet that
     * overwrites player builds — bias empty/replaceable cells in planning, not only at place time.
     */
    public static List<PlacementIntent> planPhase(int phase, int structureSize, Random random) {
        List<PlacementIntent> intents = new ArrayList<>();
        int radius = Math.max(2, structureSize / 2);
        switch (phase) {
            case 0 -> {
                for (int x = -radius; x <= radius; x++) {
                    for (int z = -radius; z <= radius; z++) {
                        intents.add(new PlacementIntent(new BlockOffset(x, 0, z), BlockKind.RESIN));
                    }
                }
            }
            case 1 -> {
                for (int x = -radius; x <= radius; x++) {
                    for (int z = -radius; z <= radius; z++) {
                        if (Math.abs(x) == radius || Math.abs(z) == radius) {
                            intents.add(new PlacementIntent(new BlockOffset(x, 1, z), BlockKind.RESIN));
                        }
                    }
                }
            }
            case 2 -> intents.add(new PlacementIntent(new BlockOffset(0, 1, 0), BlockKind.SPINY));
            case 3 -> intents.add(new PlacementIntent(new BlockOffset(0, 0, 0), BlockKind.AIR_ENTRANCE));
            case 4 -> intents.add(new PlacementIntent(new BlockOffset(1, 1, 1), BlockKind.COCOON));
            default -> {
            }
        }
        return intents;
    }
}

/**
 * Domain policy for facehugger attachment activation, host-claim gating, approach FSM, and follow range.
 */
public final class FacehuggerAttachmentPolicy {

    private FacehuggerAttachmentPolicy() {
    }

    public static boolean canActivateAttachment(
            boolean spent,
            boolean validHost,
            boolean hostClaimedBySignal,
            double distanceToTarget,
            double followRange) {
        if (spent || !validHost || hostClaimedBySignal) {
            return false;
        }
        return distanceToTarget <= followRange;
    }

    public static boolean isHostClaimedBySignal(
            boolean hostTargeted,
            boolean hostReserved,
            boolean hostAttached) {
        return hostTargeted || hostReserved || hostAttached;
    }

    public static boolean shouldBroadcastHostTargeted(boolean validTarget) {
        return validTarget;
    }

    /**
     * Another facehugger owns {@code HOST_ATTACHED} for this host (this one should stop).
     */
    public static boolean isHostAttachedByOther(
            boolean hasHostAttachedSignal,
            boolean thisFacehuggerOwnsActiveSignal) {
        if (!hasHostAttachedSignal) {
            return false;
        }
        return !thisFacehuggerOwnsActiveSignal;
    }

    public static boolean shouldContinueAttachment(
            boolean hostInvalid,
            boolean hostAttachedByOther,
            boolean attachmentInProgress) {
        if (hostInvalid || hostAttachedByOther) {
            return false;
        }
        return attachmentInProgress;
    }

    /**
     * Goal must stay active while latched even when ticks reach {@code totalDuration},
     * because entity AI evaluates {@code canContinue} before implantation processing.
     * Spent / detaching never count as in-progress (ticks are reset to 0 on spent).
     */
    public static boolean resolveAttachmentInProgress(
            AttachmentState state,
            int attachmentTicks,
            int totalDuration) {
        if (AttachmentRules.shouldProcessImplantationTick(state)) {
            return true;
        }
        if (AttachmentRules.isSpentState(state) || state == AttachmentState.DETACHING) {
            return false;
        }
        return AttachmentRules.isAttachmentInProgress(attachmentTicks, totalDuration);
    }

    public static double resolveFollowRange(double configFollowRange, double attributeFollowRange) {
        if (configFollowRange > 0) {
            return configFollowRange;
        }
        if (attributeFollowRange > 0) {
            return attributeFollowRange;
        }
        throw new IllegalStateException("Facehugger follow_range missing from entity JSON and entity attributes");
    }

    /**
     * Approach-tick decision for the attachment goal (move vs stop + FSM state).
     */
    public record ApproachTickDecision(
            AttachmentState nextState,
            boolean shouldMoveTowardHost,
            boolean shouldStopNavigation) {
    }

    public static ApproachTickDecision resolveApproachTick(
            AttachmentState currentState,
            double distanceToTarget,
            double minDistance) {
        if (currentState == AttachmentState.ATTACHING || currentState == AttachmentState.ATTACHED) {
            return new ApproachTickDecision(currentState, false, false);
        }
        AttachmentState nextState = AttachmentRules.nextApproachState(currentState, distanceToTarget, minDistance);
        if (distanceToTarget > minDistance) {
            return new ApproachTickDecision(nextState, true, false);
        }
        boolean stateChanged = nextState != currentState;
        return new ApproachTickDecision(nextState, false, stateChanged);
    }
}

            double targetX,
            double targetZ,
            float currentYaw) {
        double dx = targetX - facehuggerX;
        double dz = targetZ - facehuggerZ;
        double distance = Math.sqrt(dx * dx + dz * dz);
        if (distance < 0.1) {
            return currentYaw;
        }
        return (float) (Math.atan2(dz, dx) * 180.0 / Math.PI) - 90.0f;
    }

            this.hostType = hostType;

    private static void registerTierOneMobVariants() {

        if (!MobEntityConfigLoader.getInstance().isLoaded()) {

            ModLogger.warn(

                    "MobEntityConfigLoader not loaded — specialized xenomorph variants unavailable until configs load");

            return;

        }



        for (String mob : MobVariantManifest.mobNames()) {

            LoadedMobVariantConfig config = MobEntityConfigLoader.getConfigForMob(mob);

            if (config != null) {

                registerVariant(fromMobEntityConfig(mob, config));

            } else {

                ModLogger.warn("No mob entity config for '{}' — skipping specialized variant", mob);

            }

        }

    }



    private static SpecializedXenomorphConfig fromMobEntityConfig(String mob, LoadedMobVariantConfig config) {

        String variantName = mob + "_xenomorph";

        HostType hostType = MobEntityConfigLoader.hostTypeFromCatalogKey(config.hostType);

        List<String> abilities = config.specialAbilities.isEmpty()

                ? deriveSpecialAbilities(config)

                : new ArrayList<>(config.specialAbilities);



        return new SpecializedXenomorphConfig(

                variantName,

                hostType,

                "drone",

                abilities,

                buildVariantProperties(config),

                config.description != null && !config.description.isEmpty()

                        ? config.description

                        : mob + "-hosted xenomorph variant");

    }



    private static List<String> deriveSpecialAbilities(LoadedMobVariantConfig config) {

        List<String> abilities = new ArrayList<>();

        if (config.climbingAbility) {

            abilities.add("wall_climbing");

        }

        if (config.webAbilities) {

            abilities.add("web_weaving");

        }

        if (config.infectionSpread) {

            abilities.add("explosive_ability");

        }

        if (config.packBehavior) {

            abilities.add("pack_hunting");

        }

        if (config.enhancedSenses) {

            abilities.add("enhanced_senses");

        }

        if (config.decayResistance) {

            abilities.add("decay_resistance");

        }

        if (config.metalConsumption) {

            abilities.add("metal_consumption");

        }

        return abilities;

    }



    private static Map<String, Object> buildVariantProperties(LoadedMobVariantConfig config) {

        Map<String, Object> props = new HashMap<>();

        props.put("health_multiplier", config.healthMultiplier);

        props.put("speed_multiplier", config.speedMultiplier);

        props.put("damage_multiplier", config.damageMultiplier);

        props.put("aggression_bonus", config.aggressionBonus);

        props.put("stealth_bonus", config.stealthBonus);



        if (config.climbingAbility) {

            props.put("climbing_ability", true);

        }

        if (config.webAbilities) {

            props.put("web_weaving", true);

        }

        if (config.infectionSpread) {

            props.put("stealth_ability", true);

import com.biocraft.alien.layer_0_domain.level_0.spatial.BlockPosRef;
import java.util.function.Predicate;

/**
 * Exhaustive spawn-location search loop (O(n³); suitability probes supplied by L3).
 */
public final class SpawnLocationSearchPolicy {

    private SpawnLocationSearchPolicy() {}

    public static BlockPosRef findFirstSuitable(
            BlockPosRef origin, int searchRadius, int maxHeight, Predicate<BlockPosRef> suitabilityProbe) {
        if (origin == null || suitabilityProbe == null || searchRadius <= 0 || maxHeight <= 0) {
            return null;
        }

        for (int x = -searchRadius; x <= searchRadius; x++) {
            for (int z = -searchRadius; z <= searchRadius; z++) {
                for (int y = -maxHeight; y <= maxHeight; y++) {
                    BlockPosRef candidate = origin.offset(x, y, z);
                    if (suitabilityProbe.test(candidate)) {
                        return candidate;
                    }
                }
            }
        }
        return null;
    }
}

import java.util.Objects;

import java.util.UUID;

import com.biocraft.alien.layer_0_domain.level_0.lab.ContainmentOccupancySnapshot;
import com.biocraft.alien.layer_0_domain.level_0.validation.ValidationResult;
import com.biocraft.alien.layer_0_domain.level_1.entity.LivingEntityRef;
import com.biocraft.alien.layer_2_systems.level_0.containment.ContainmentOccupancyPort;

/**
 * Identity-based occupancy policy. This service never resolves entities or caches state.
 */
public final class ContainmentOccupancyService {

    private ContainmentOccupancyService() {
    }

    public static ValidationResult admit(ContainmentOccupancyPort port, LivingEntityRef occupant) {
        Objects.requireNonNull(port, "port");
        if (occupant == null || occupant.entityId() == null || occupant.world() == null) {
            return ValidationResult.failure("occupant identity is required");
        }
        if (!port.isStructureFormed()) {
            return ValidationResult.failure("containment structure is not formed");
        }
        if (port.occupantEntityId().isPresent()) {
            return ValidationResult.failure("containment is already occupied");
        }
        if (!Objects.equals(port.controllerWorld(), occupant.world())) {
            return ValidationResult.failure("occupant belongs to a different world");
        }

        port.setOccupantEntityId(occupant.entityId());
        return ValidationResult.success();
    }

    public static Optional<UUID> release(ContainmentOccupancyPort port) {
        Objects.requireNonNull(port, "port");
        Optional<UUID> occupant = port.occupantEntityId();
        if (occupant.isPresent()) {
            port.clearOccupantEntityId();
        }
        return occupant;
    }

    public static void clearOnBreach(ContainmentOccupancyPort port) {
        Objects.requireNonNull(port, "port");
        if (port.occupantEntityId().isPresent()) {
            port.clearOccupantEntityId();
        }
    }

    /**
     * Returns an uncached snapshot without mutating containment or world state.
     */
    public static ContainmentOccupancySnapshot query(ContainmentOccupancyPort port) {
        Objects.requireNonNull(port, "port");
        return new ContainmentOccupancySnapshot(port.isStructureFormed(), port.occupantEntityId());
    }
}

/**
 * Pure containment pen multiblock orchestration. Block I/O via {@link ContainmentPenPort}.
 */
public final class ContainmentPenWorkflow {

    private static final String CONTROLLER_BLOCK = "containment_pen_controller";

    private ContainmentPenWorkflow() {
    }

    public static boolean processPeriodicTick(ContainmentPenPort port, long gameTime) {
        if (gameTime % validationIntervalTicks() != 0) {
            return false;
        }
        if (port.isFormed() && !validateStructure(port)) {
            port.logDebug("Containment pen structure broken at " + port.controllerPosition());
            breakStructure(port);
            return true;
        }
        return false;
    }

    public static boolean validateStructure(ContainmentPenPort port) {
        port.logDebug("Validating containment pen structure at " + port.controllerPosition());

        if (!port.isControllerBlock()) {
            port.setStructureState(port.structureState().transitionOnValidate(false));
            port.markDirty();
            return false;
        }

        if (port.isFormed()) {
            boolean intact = port.formedStructureIntact();
            if (!intact) {
                port.logDebug("Formed containment pen shells no longer intact at " + port.controllerPosition());
            }
            return intact;
        }

        ContainmentPenBlueprint.ContainmentPenValidationResult result = ContainmentPenBlueprint.validate(
                port::isOccupied);
        port.applyValidationResult(result);
        return result.isValid();
    }

    public static void breakStructure(ContainmentPenPort port) {
        port.logInfo("Breaking containment pen structure at " + port.controllerPosition());
        port.dissolveFormedStructure();
        port.setStructureBlocks(List.of());
        port.setStructureState(port.structureState().transitionOnBreak());
        port.markDirty();
    }

    public static List<BlockPosRef> offsetsToWorld(BlockPosRef center, List<ContainmentPenBlueprint.BlockOffset> offsets) {
        List<BlockPosRef> world = new ArrayList<>(offsets.size());
        for (ContainmentPenBlueprint.BlockOffset offset : offsets) {
            world.add(center.offset(offset.x(), offset.y(), offset.z()));
        }
        return world;
    }

    private static int validationIntervalTicks() {
        LoadedBlockVisualConfig config = BlockConfigLoader.getConfig(CONTROLLER_BLOCK);
        if (config == null || config.validationIntervalTicks <= 0) {
            throw new ConfigContractException("ContainmentPenWorkflow",
                    "/data/biocraft_alien/blocks/" + CONTROLLER_BLOCK + ".json",
                    "gameplay.validation_interval_ticks", "required positive integer");
        }
        return config.validationIntervalTicks;
    }
}

/**
 * Focused manager for gestation effect logic.
 * Timings and damage come from {@code data/biocraft_alien/effects/gestation.json}.
 *
 * <p>TODO(GAMEPLAY-GESTATION-001): Accepted debt — JSON owns timings; residual Java: player duration *= 3, log % 200, min damage floor.
 *
 * FILE STATUS: STABLE
 */
public class GestationManager {

    public static void applyGestationEffect(LivingEntityRef host) {
        if (EntityValidationHelper.isInvalid(host, BoundaryServices.entityQuery()) || !BoundaryServices.isInitialized()) {
            ModLogger.info("Skipping gestation application - host is invalid or boundary not ready");
            return;
        }

        if (BoundaryServices.effects().hasEffect(host, DomainEffectKey.GESTATION)) {
            ModLogger.info("Skipping gestation application - host already has effect");
            return;
        }

        int gestationDuration = getGestationDuration();
        if (BoundaryServices.entityQuery().isPlayer(host)) {
            gestationDuration *= 3;
        }

        BoundaryServices.effects().apply(host, DomainEffectKey.GESTATION, gestationDuration, 0,
                new EffectApplyOptions(false, true, true));

        if (BoundaryServices.effects().hasEffect(host, DomainEffectKey.GESTATION)) {
            ModLogger.info("Gestation effect applied to "
                    + BoundaryServices.presentation().displayName(host)
                    + " for " + gestationDuration + " ticks");
        } else {
            ModLogger.error("Failed to apply gestation effect to "
                    + BoundaryServices.presentation().displayName(host));
        }
    }

    public static boolean handleGestationTick(LivingEntityRef host, int timeRemaining) {
        if (!BoundaryServices.isInitialized()) {
            return false;
        }

        if (timeRemaining < 0) {
            ModLogger.error("Invalid gestation duration: " + timeRemaining + " ticks for "
                    + BoundaryServices.presentation().displayName(host));
            return false;
        }

        if (timeRemaining % 200 == 0) {
            ModLogger.info("Gestation progress for "
                    + BoundaryServices.presentation().displayName(host)
                    + ": " + timeRemaining + " ticks remaining");
        }

        int warningPhaseStart = getWarningPhaseStart();
        int finalPhaseStart = getFinalPhaseStart();

        if (timeRemaining == warningPhaseStart) {
            ModLogger.info("Gestation warning phase starting for "
                    + BoundaryServices.presentation().displayName(host)
                    + " - " + timeRemaining + " ticks remaining");
        }

        if (timeRemaining == finalPhaseStart) {
            ModLogger.info("Gestation final phase starting for "
                    + BoundaryServices.presentation().displayName(host)
                    + " - " + timeRemaining + " ticks remaining");
        }

        if (timeRemaining <= warningPhaseStart && timeRemaining > finalPhaseStart) {
            handleWarningPhase(host, timeRemaining);
        }

        if (timeRemaining <= finalPhaseStart) {
            handleFinalPhase(host);
            return false;
        }

        return true;
    }

    private static void handleWarningPhase(LivingEntityRef host, int timeRemaining) {
        int damageInterval = getWarningDamageInterval();
        // damage_amount is a fraction of max HP (e.g. 0.05 = 5%).
        float damageFraction = getWarningDamageAmount();
        int particleCount = getWarningParticleCount();
        float particleSpread = getWarningParticleSpread();

        if (timeRemaining % damageInterval == 0) {
            float maxHp = BoundaryServices.presentation().maxHealth(host);
            float damage = Math.max(0.5f, maxHp * damageFraction);
            BoundaryServices.presentation().applyMagicDamage(host, damage);
        }

        BoundaryServices.presentation().spawnDamageIndicatorParticles(host, particleCount, particleSpread);
    }

    private static void handleFinalPhase(LivingEntityRef host) {
        ModLogger.info("Gestation complete! Chestburster emerging from "
                + BoundaryServices.presentation().displayName(host));

        spawnChestburster(host);

        if (isFinalPhaseInstantKill()) {
            ModLogger.info("Killing host " + BoundaryServices.presentation().displayName(host)
                    + " (instant_kill)");
            BoundaryServices.presentation().kill(host);
        } else {
            float damageFraction = getFinalDamageAmount();
            float maxHp = BoundaryServices.presentation().maxHealth(host);
            float damage = Math.max(1.0f, maxHp * damageFraction);
            ModLogger.info("Killing host " + BoundaryServices.presentation().displayName(host)
                    + " with " + damage + " damage (" + damageFraction + " of max HP)");
            BoundaryServices.presentation().applyMagicDamage(host, damage);
        }

        ModLogger.info("Gestation process completed for "
                + BoundaryServices.presentation().displayName(host));
    }

    private static void spawnChestburster(LivingEntityRef host) {
        if (!BoundaryServices.isInitialized()) {
            ModLogger.warn("BoundaryServices not initialized — cannot spawn chestburster");
            return;
        }

        try {
            BoundaryServices.entitySpawn().spawnXenomorph(
                    BoundaryServices.presentation().spawnContextAt(host),
                    XenomorphType.CHESTBURSTER);

            BoundaryServices.presentation().spawnExplosionParticles(
                    host, getExplosionParticleCount(), getExplosionParticleSpread());

            ModLogger.info("Chestburster spawned successfully for host "
                    + BoundaryServices.presentation().displayName(host));
        } catch (Exception e) {
            ModLogger.error("Failed to spawn chestburster: " + e.getMessage(), e);
        }
    }

    public static boolean isGestating(LivingEntityRef entity) {
        return entity != null && BoundaryServices.isInitialized()
                && BoundaryServices.effects().hasEffect(entity, DomainEffectKey.GESTATION);
    }

    public static int getRemainingGestationTime(LivingEntityRef entity) {
        if (!isGestating(entity)) {
            return 0;
        }
        return BoundaryServices.effects().getRemainingDuration(entity, DomainEffectKey.GESTATION);
    }

    private static EffectConfig gestationConfig() {
        return EffectConfigLoader.getConfig("gestation");
    }

    private static int getGestationDuration() {
        EffectConfig.DurationConfig duration = gestationConfig().duration;
        if (duration == null || duration.durationTicks <= 0) {
            throw new ConfigContractException("gestation", "duration.duration_ticks", "required positive");
        }
        return duration.durationTicks;
    }

    private static int getWarningPhaseStart() {
        EffectConfig.WarningPhaseConfig warning = requireWarningPhase();
        return warning.startTicks;
    }

    private static int getFinalPhaseStart() {
        EffectConfig.FinalPhaseConfig finalPhase = requireFinalPhase();
        return finalPhase.startTicks;
    }

    private static int getWarningDamageInterval() {
        return requireWarningPhase().damageInterval;
    }

    private static float getWarningDamageAmount() {
        return (float) requireWarningPhase().damageAmount;
    }

    private static int getWarningParticleCount() {
        EffectConfig.VisualEffectsConfig visual = gestationConfig().visualEffects;
        if (visual == null) {
            throw new ConfigContractException("gestation", "visual_effects", "required");
        }
        return visual.warningParticleCount;
    }

    private static float getWarningParticleSpread() {
        EffectConfig.VisualEffectsConfig visual = gestationConfig().visualEffects;
        if (visual == null) {
            throw new ConfigContractException("gestation", "visual_effects", "required");
        }
        return (float) visual.warningParticleSpread;
    }

    private static float getFinalDamageAmount() {

    private static boolean isFinalPhaseInstantKill() {

    private static int getExplosionParticleCount() {

package com.biocraft.alien.layer_2_systems.level_1.hive;

import com.biocraft.alien.layer_0_domain.level_0.spatial.BlockPosRef;
import com.biocraft.alien.layer_2_systems.level_0.hive.NestBuildingPort;
import com.biocraft.alien.layer_2_systems.level_0.hive.NestBuildingTiming;

/**
 * Orchestration workflow for nest building lifecycle.
 * Encapsulates timing, phase progression, and hive expansion logic.
 * 
 * Boundaries:
 * - Does NOT import Minecraft types
 * - Does NOT manage territory (delegates to port)
 * - Does NOT perform block placement (delegates to port)
 * - Manages timing and phase progression state
 *
 * <p>TODO(FEATURE-NEST-002): delay nest claim until after a short idle/wander window
 * (or explicit “ready to nest” signal) instead of validating/claiming as soon as cooldown
 * allows — prefer choosing an ideal open site after that delay.
 */
public final class NestBuildingWorkflow {
    private final NestBuildingPort port;
    private final NestBuildingTiming timing;
    private int buildingProgress;
    private int hiveBuildingProgress;
    private int phaseDelayTicks;
    private int cooldownTicks;
    private boolean expandingToHive;

    public NestBuildingWorkflow(NestBuildingPort port, NestBuildingTiming timing) {
        this.port = port;
        this.timing = timing;
    }

    /**
     * Advance the nest building workflow by one tick.
     * Handles cooldown, phase delays, nest progression, and hive expansion.
     */
    public void tick() {
        // Cooldown period after nest creation
        if (this.cooldownTicks > 0) {
            --this.cooldownTicks;
            return;
        }
        
        // Delay between phases
        if (this.phaseDelayTicks > 0) {
            --this.phaseDelayTicks;
            return;
        }

        // Claim nest center before structure phases (territory exclusivity via port)
        // TODO(FEATURE-NEST-002): score open-area candidates; do not always claim queen feet.
        if (this.port.currentNestCenter() == null) {
            if (!this.port.validateNestLocation()) {
                // Stay quiet most ticks; nest claim failures are common in invalid terrain.
                return;
            }
            this.port.registerTerritory(this.port.queenPosition(), false);
        }

        // Build nest structure phases
        if (this.buildingProgress < this.timing.nestBuildingPhases()) {
            if (this.port.buildNestStructurePhase(this.buildingProgress)) {
                ++this.buildingProgress;
                this.phaseDelayTicks = this.timing.phaseBuildDelayTicks();
            }
            return;
        }
        
        // Check if ready to expand to hive
        if (!this.expandingToHive && this.port.checkTerritoryProgression()) {
            this.expandingToHive = true;
            this.hiveBuildingProgress = 0;
        }
        
        // Build hive structure phases
        if (this.expandingToHive && this.hiveBuildingProgress < this.timing.hiveBuildingPhases()) {
            if (this.port.buildHiveStructurePhase(this.hiveBuildingProgress)) {
                ++this.hiveBuildingProgress;
                this.phaseDelayTicks = this.timing.phaseBuildDelayTicks();
            }
            if (this.hiveBuildingProgress >= this.timing.hiveBuildingPhases()) {
                this.port.onHiveStructureComplete();
            }
        }
    }

    /**
     * Restore workflow state from saved data.
     */
    public void restoreFromSave(BlockPosRef nestCenter, int savedBuildingProgress, int savedHiveProgress, boolean savedExpanding) {
        this.port.restoreNestState(nestCenter, savedBuildingProgress, savedHiveProgress, savedExpanding);
        this.buildingProgress = savedBuildingProgress;
        this.hiveBuildingProgress = savedHiveProgress;
        this.expandingToHive = savedExpanding;
        this.cooldownTicks = this.timing.nestCreationCooldownTicks();
    }

    public int getBuildingProgress() {
        return this.buildingProgress;
    }

    public boolean isNestComplete() {
        return this.buildingProgress >= this.timing.nestBuildingPhases();
    }

    public int getHiveBuildingProgress() {
        return this.hiveBuildingProgress;
    }

    public boolean isExpandingToHive() {
        return this.expandingToHive;
    }

    public boolean isHiveExpansionComplete() {
        return this.expandingToHive && this.hiveBuildingProgress >= this.timing.hiveBuildingPhases();
    }
}

                ref -> LifecycleConfig.getDefaultFoodTimerReductionTicks());
    }

    private static void advanceMaturationTimer(@Nonnull LivingEntityRef entity) {
        EntityConfig config = maturationConfig(entity);
        if (config == null || config.maturationTime <= 0) {
            return;
        }
        int timer = getMaturationTimer(entity);
        if (timer < config.maturationTime) {
            setMaturationTimer(entity, timer + 1);
        }
    }

    private static void reduceMaturationTimerGeneric(@Nonnull LivingEntityRef entity, int reduction) {
        if (reduction <= 0) {
            return;
        }
        int timer = getMaturationTimer(entity);
        setMaturationTimer(entity, Math.max(0, timer - reduction));
    }

    private static boolean shouldPrioritizeFoodHuntingGeneric(@Nonnull LivingEntityRef entity) {
        return getMaturationUrgencyGeneric(entity) < LifecycleConfig.getFoodHuntUrgencyThreshold();
    }

    private static double getMaturationUrgencyGeneric(@Nonnull LivingEntityRef entity) {
        EntityConfig config = maturationConfig(entity);
        if (config == null || config.maturationTime <= 0) {
            return 1.0;
        }
        int timer = getMaturationTimer(entity);
        return Math.min(1.0, (double) timer / config.maturationTime);
    }

    @Nullable
    private static EntityConfig maturationConfig(@Nonnull LivingEntityRef entity) {
        if (shouldSkipEntity(entity)) {
            return null;
        }
        XenomorphType type = resolveType(entity);
        if (type == null) {
            return null;
        }
        EntityConfig config = EntityConfigStorage.getConfig(type);
        if (config == null || !config.canMature) {
            return null;
        }
        return config;
    }

    private static void registerType(XenomorphType type,
                                     MaturationHandler processor,
                                     ObjIntConsumer<LivingEntityRef> timerReducer,
                                     Function<LivingEntityRef, Boolean> foodHunting,
                                     ToDoubleFunction<LivingEntityRef> urgency,
                                     ToIntFunction<LivingEntityRef> timerReduction) {
        PROCESS_REGISTRY.put(type, processor);
        TIMER_GETTER_REGISTRY.put(type, MaturationSystem::readMaturationTimer);
        TIMER_SETTER_REGISTRY.put(type, MaturationSystem::writeMaturationTimer);
        TIMER_REDUCER_REGISTRY.put(type, timerReducer);
        FOOD_HUNTING_REGISTRY.put(type, foodHunting);
        URGENCY_REGISTRY.put(type, urgency);
        TIMER_REDUCTION_REGISTRY.put(type, timerReduction);
    }

    public static void registerEntityType(@Nonnull XenomorphType type,
                                          @Nonnull MaturationHandler processor,
                                          @Nonnull ToIntFunction<LivingEntityRef> timerGetter,
                                          @Nonnull ObjIntConsumer<LivingEntityRef> timerSetter,
                                          @Nonnull ObjIntConsumer<LivingEntityRef> timerReducer,
                                          @Nonnull Function<LivingEntityRef, Boolean> foodHunting,
                                          @Nonnull ToDoubleFunction<LivingEntityRef> urgency,
                                          @Nonnull ToIntFunction<LivingEntityRef> timerReduction) {
        PROCESS_REGISTRY.put(type, processor);
        TIMER_GETTER_REGISTRY.put(type, timerGetter);
        TIMER_SETTER_REGISTRY.put(type, timerSetter);
        TIMER_REDUCER_REGISTRY.put(type, timerReducer);
        FOOD_HUNTING_REGISTRY.put(type, foodHunting);
        URGENCY_REGISTRY.put(type, urgency);
        TIMER_REDUCTION_REGISTRY.put(type, timerReduction);
    }

    public static void processMaturation(@Nonnull LivingEntityRef entity) {
        if (shouldSkipEntity(entity)) {
            return;
        }
        MaturationHandler handler = PROCESS_REGISTRY.get(resolveType(entity));
        if (handler != null) {
            handler.process(entity);
        }
    }

    public static int resolveMaturationTimer(@Nonnull LivingEntityRef entity) {
        return readMaturationTimer(entity);
    }

    public static void resolveSetMaturationTimer(@Nonnull LivingEntityRef entity, int timer) {
        writeMaturationTimer(entity, timer);
    }

    private static int readMaturationTimer(@Nonnull LivingEntityRef entity) {
        if (MaturationServices.isInitialized()) {
            return MaturationServices.maturation().getMaturationTimer(entity);
        }
        return 0;
    }

    private static void writeMaturationTimer(@Nonnull LivingEntityRef entity, int timer) {
        if (MaturationServices.isInitialized()) {
            MaturationServices.maturation().setMaturationTimer(entity, timer);
        }
    }

    public static int getMaturationTimer(@Nonnull LivingEntityRef entity) {
        if (shouldSkipEntity(entity)) {
            return 0;
        }
        ToIntFunction<LivingEntityRef> getter = TIMER_GETTER_REGISTRY.get(resolveType(entity));
        return getter != null ? getter.applyAsInt(entity) : 0;
    }

    public static void setMaturationTimer(@Nonnull LivingEntityRef entity, int timer) {
        if (shouldSkipEntity(entity)) {
            return;
        }
        ObjIntConsumer<LivingEntityRef> setter = TIMER_SETTER_REGISTRY.get(resolveType(entity));
        if (setter != null) {
            setter.accept(entity, timer);
        }
    }

    public static void reduceMaturationTimer(@Nonnull LivingEntityRef entity, int reduction) {
        if (shouldSkipEntity(entity)) {
            return;
        }
        ObjIntConsumer<LivingEntityRef> reducer = TIMER_REDUCER_REGISTRY.get(resolveType(entity));
        if (reducer != null) {
            reducer.accept(entity, reduction);
        }
    }

    public static boolean shouldPrioritizeFoodHunting(@Nonnull LivingEntityRef entity) {
        if (shouldSkipEntity(entity)) {
            return false;
        }
        Function<LivingEntityRef, Boolean> handler = FOOD_HUNTING_REGISTRY.get(resolveType(entity));
        return handler != null && handler.apply(entity);
    }

    public static double getMaturationUrgency(@Nonnull LivingEntityRef entity) {
        if (shouldSkipEntity(entity)) {
            return 0.0;
        }
        ToDoubleFunction<LivingEntityRef> handler = URGENCY_REGISTRY.get(resolveType(entity));
        return handler != null ? handler.applyAsDouble(entity) : 0.0;
    }

    public static int getFoodTimerReduction(@Nonnull LivingEntityRef entity) {
        if (shouldSkipEntity(entity)) {
            return 0;
        }
        ToIntFunction<LivingEntityRef> handler = TIMER_REDUCTION_REGISTRY.get(resolveType(entity));
        return handler != null ? handler.applyAsInt(entity) : 0;
    }

    private static boolean shouldSkipEntity(@Nonnull LivingEntityRef entity) {
        return !BoundaryServices.isInitialized()
                || EntityValidationHelper.isInvalid(entity, BoundaryServices.entityQuery());
    }

    @Nullable
    private static XenomorphType resolveType(@Nonnull LivingEntityRef entity) {
        if (!BoundaryServices.isInitialized()) {
            return null;
        }
        return BoundaryServices.formQuery().resolveForm(entity);
    }

    static {
        registerMaturationHandlers();
    }
}

/**
 * Produces observation facts only. Consumers decide relevance and ignore filters.
 */
public final class ObservationService {

    private ObservationService() {
    }

    /**
     * Returns snapshots for all facts in the port window (unfiltered).
     */
    public static List<ObservationSnapshot> observe(ObservationPort port) {
        Objects.requireNonNull(port, "port");
        ObservationInstant instant = Objects.requireNonNull(port.currentInstant(), "instant");
        List<ObservationPort.ObservationFact> facts = Objects.requireNonNull(port.facts(), "facts");
        List<ObservationSnapshot> snapshots = new ArrayList<>(facts.size());
        for (ObservationPort.ObservationFact fact : facts) {
            Objects.requireNonNull(fact, "fact");
            snapshots.add(new ObservationSnapshot(fact.entity(), instant, fact.movement()));
        }
        return List.copyOf(snapshots);
    }

    /**
     * Returns the snapshot for a specific entity if present in the current facts.
     */
    public static Optional<ObservationSnapshot> observe(ObservationPort port, LivingEntityRef entity) {
        Objects.requireNonNull(entity, "entity");
        return observe(port).stream()
                .filter(snapshot -> entity.equals(snapshot.entity()))
                .findFirst();
    }
}

package com.biocraft.alien.layer_2_systems.level_1.ovomorph;

import com.biocraft.alien.layer_2_systems.level_0.ovomorph.OvomorphHatchPort;
/**
 * Orchestration workflow for ovomorph hatching lifecycle.
 * Encapsulates timing, proximity requirements, and hatch trigger logic.
 *
 * Static utility class with mutable state record.
 * Each ovomorph entity maintains its own State instance.
 *
 * Boundaries:
 * - Does NOT import Minecraft types
 * - Does NOT perform entity spawning (delegates to port)
 * - Does NOT validate hosts (delegates to port)
 * - Manages hatching timer and state progression
 */
public final class OvomorphHatchWorkflow {

    private OvomorphHatchWorkflow() {
    }

    /**
     * Initialize hatching state when hatching begins.
     *
     * @param port the hatching port interface
     * @param state the state to initialize
     */
    public static void onHatchingStarted(OvomorphHatchPort port, State state) {
        state.startTick = -1;
        state.elapsedTicks = 0;
    }

    /**
     * Process one tick of the hatching lifecycle.
     * Checks proximity requirements, advances timer, and triggers hatch when ready.
     *
     * @param port the hatching port interface
     * @param state the current hatching state
     * @param currentTick the server tick count
     */
    public static void processTick(OvomorphHatchPort port, State state, int currentTick) {
        // Initialize start tick on first call
        if (state.startTick < 0) {
            state.startTick = currentTick;
        }

        // Calculate elapsed time
        state.elapsedTicks = currentTick - state.startTick;

        // Cancel if proximity required but no host nearby
        if (port.isProximityRequired() && !port.isViableHostNearby()) {
            port.cancelHatching();
            return;
        }

        // Update state
        state.hatchTimer = state.elapsedTicks;
        state.wasHostNearby = port.isViableHostNearby();

        // Execute hatch if time threshold reached
        if (state.elapsedTicks >= port.hatchTime()) {
            port.executeHatch();
        }
    }

    /**
     * Mutable state for ovomorph hatching process.
     * Each ovomorph instance maintains one State throughout its lifetime.
     */
    public static final class State {
        public int hatchTimer;
        public boolean wasHostNearby;
        private int startTick = -1;
        private int elapsedTicks;
    }
}

    public static void applyHostImmobilization(LivingEntityRef host) {
        if (EntityValidationHelper.isInvalid(host, BoundaryServices.entityQuery()) || !BoundaryServices.isInitialized()) {
            return;
        }

        String entityName = BoundaryServices.entityQuery().encodeId(host)
                .map(HostEffectManager::registryPath)
                .orElse("");
        LoadedHostEffectProfile profile = resolveHostEffectProfile(entityName);
        HostType hostType = MobHostRegistry.getHostType(entityName);
        if (hostType == null) {
            // TODO(BUG-HOST-002): Accepted Debt — fail closed for biocraft_alien:human (no generic HostType fallback).
            hostType = HostType.LIVING_BIOLOGICAL;
        }

        if (!profile.canBeImmobilized) {
            logUnusualHost(host, hostType, profile.profileId, "Cannot be immobilized");
            return;
        }

        applyVisualEffects(host, profile.visualEffects);
        applyMovementEffects(host, profile.movementEffects);

        // Config-driven condition list (JSON attachment_conditions / paralysis) — engine maps via port.
        List<AttachmentCondition> conditions = AttachmentConfigHelper.getAttachmentConditions();
        for (AttachmentCondition condition : conditions) {
            BoundaryServices.presentation().applyAttachmentCondition(host, condition);
        }

        if (profile.requiresPlayerSpecificEffects && BoundaryServices.entityQuery().isPlayer(host)) {
            BoundaryServices.presentation().applyPlayerAttachmentPose(host);
        }

        if (BoundaryServices.presentation().tickCount(host) % 200 == 0) {
            ModLogger.info("Host " + BoundaryServices.presentation().displayName(host)
                    + " (" + hostType.getKey() + "/" + profile.profileId
                    + ") is being facehugged - immobilized");
        }
    }

    private static LoadedHostEffectProfile resolveHostEffectProfile(String entityName) {
        String profileId = MobHostRegistry.getDnaProfileId(entityName);
        if (profileId == null || profileId.isBlank()) {
            // TODO(BUG-HOST-002): Accepted Debt — fail closed for biocraft_alien:human (no generic profile fallback).
            ModLogger.warn("Unknown entity type '" + entityName
                    + "' - defaulting to host-effect profile "
                    + HostEffectProfileNames.BASELINE_BIOLOGICAL);
            profileId = HostEffectProfileNames.BASELINE_BIOLOGICAL;
        }
        return HostEffectProfileStorage.get(profileId);
    }

                continue;

            EffectApplyOptions options = new EffectApplyOptions(
                    false, effectConfig.showParticles, effectConfig.showParticles);
            if (effectPort.hasStandardEffect(host, effectConfig.standardEffect)) {
                continue;

            effectPort.applyStandard(host, effectConfig.standardEffect, effectConfig.duration,
                    effectConfig.amplifier, options);
        }
    }

/**
 * Owns research invariants. Validates then appends via {@link ResearchPort}.
 *
 * <p>Does not analyze biology, mutate entities, update records in place,
 * auto-subscribe to observation, or trigger gameplay.
 */
public final class ResearchService {

    private ResearchService() {
    }

    /**
     * Validates fact keys and appends an immutable record. Rejects duplicate ids
     * (append-only — corrections must use a new record id).
     */
    public static ResearchResult record(ResearchPort port, ResearchRecord researchRecord) {
        Objects.requireNonNull(port, "port");
        Objects.requireNonNull(researchRecord, "researchRecord");

        for (ResearchFact fact : researchRecord.facts()) {
            if (!ResearchFactKeys.isAllowed(fact.key())) {
                return ResearchResult.rejected("illegal research fact key: " + fact.key());
            }
        }

        boolean duplicate = port.records().stream()
                .anyMatch(existing -> existing.id().equals(researchRecord.id()));
        if (duplicate) {
            return ResearchResult.rejected("research record already exists (append-only)");
        }

        port.appendRecord(researchRecord);
        return ResearchResult.ok();
    }

    /**
     * Collects a custody sample. Remains queryable even if the source entity later disappears.
     * Also appends a SAMPLE-category research record for history.
     */
    public static SampleRecord collectSample(
            ResearchPort port,
            LivingEntityRef source,
            SampleKind kind) {
        Objects.requireNonNull(port, "port");
        Objects.requireNonNull(source, "source");
        Objects.requireNonNull(kind, "kind");

        ResearchInstant instant = Objects.requireNonNull(port.currentInstant(), "instant");
        SampleRecord sample = new SampleRecord(SampleId.random(), source, instant, kind);
        port.appendSample(sample);

        ResearchRecord companion = new ResearchRecord(
                ResearchRecordId.random(),
                Optional.of(source),
                ResearchCategory.SAMPLE,
                instant,
                List.of(new ResearchFact(ResearchFactKeys.SAMPLE_KIND, kind.name())));
        ResearchResult companionResult = record(port, companion);
        if (!companionResult.accepted()) {
            throw new IllegalStateException("sample companion record rejected: " + companionResult.reason());
        }
        return sample;
    }

    public static List<ResearchRecord> queryBySubject(ResearchPort port, LivingEntityRef subject) {
        Objects.requireNonNull(port, "port");
        Objects.requireNonNull(subject, "subject");
        List<ResearchRecord> matches = new ArrayList<>();
        for (ResearchRecord researchRecord : port.records()) {
            if (researchRecord.subject().isPresent() && subject.equals(researchRecord.subject().get())) {
                matches.add(researchRecord);
            }
        }
        return List.copyOf(matches);
    }

    public static List<ResearchRecord> queryByCategory(ResearchPort port, ResearchCategory category) {
        Objects.requireNonNull(port, "port");
        Objects.requireNonNull(category, "category");
        List<ResearchRecord> matches = new ArrayList<>();
        for (ResearchRecord researchRecord : port.records()) {
            if (category.equals(researchRecord.category())) {
                matches.add(researchRecord);
            }
        }
        return List.copyOf(matches);
    }

    /**
     * Full append-ordered history (storage order).
     */
    public static List<ResearchRecord> history(ResearchPort port) {
        Objects.requireNonNull(port, "port");
        return List.copyOf(port.records());
    }

    public static List<SampleRecord> querySamples(ResearchPort port) {

/**
 * Hybrid AI signal coordination store (domain protocol + world position binding).
 */
public class AISignalSystem {

    private static final Map<AISignalType, List<AISignalData>> activeSignals = new ConcurrentHashMap<>();
    private static final Map<UUID, List<AISignalData>> entitySignals = new ConcurrentHashMap<>();

    private AISignalSystem() {}

    public static AISignalData broadcastSignal(
            AISignalType type,
            LivingEntityRef sourceEntity,
            BlockPosRef position,
            double radius,
            long gameTime,
            int durationTicks,
            String sourceLabel) {
        if (sourceEntity == null || position == null) {
            ModLogger.warn("AISignalSystem", "⚠️ Cannot broadcast signal - source or position is null");
            return null;
        }

        AISignalData signal = new AISignalData(
                type, sourceEntity.entityId(), position, radius, gameTime, durationTicks);
        activeSignals.computeIfAbsent(type, k -> new ArrayList<>()).add(signal);
        entitySignals.computeIfAbsent(sourceEntity.entityId(), k -> new ArrayList<>()).add(signal);

        ModLogger.debug(
                "AISignalSystem",
                "📡 Broadcast signal: " + type + " from " + sourceLabel + " at "
                        + position + " (radius: " + radius + ", duration: " + durationTicks + " ticks)");

        return signal;
    }

    public static AISignalData getSignal(AISignalType type, BlockPosRef position, double maxDistance, long currentTick) {
        List<AISignalData> signals = activeSignals.get(type);
        if (signals == null || signals.isEmpty()) {
            return null;
        }

        AISignalData strongest = null;
        double strongestIntensity = 0.0;

        for (AISignalData signal : signals) {
            if (signal.isExpired(currentTick)) {
                continue;
            }
            double intensity = signal.getIntensity(position);
            if (intensity > strongestIntensity && intensity > 0.0) {
                strongest = signal;
                strongestIntensity = intensity;
            }
        }
        return strongest;
    }

    public static boolean hasSignalForEntity(AISignalType type, UUID entityId, long currentTick) {
        List<AISignalData> signals = activeSignals.get(type);
        if (signals == null || signals.isEmpty()) {
            return false;
        }

        for (AISignalData signal : signals) {
            if (signal.isExpired(currentTick)) {
                continue;
            }
            Object targetId = signal.getMetadata().get("targetEntityId");
            if (targetId instanceof UUID && targetId.equals(entityId)) {
                return true;
            }
        }
        return false;
    }

    public static boolean hasSignalForEntityExcludingSource(
            AISignalType type, UUID entityId, UUID excludeSourceId, long currentTick) {
        List<AISignalData> signals = activeSignals.get(type);
        if (signals == null || signals.isEmpty()) {
            return false;
        }

        for (AISignalData signal : signals) {
            if (signal.isExpired(currentTick)) {
                continue;
            }
            if (excludeSourceId != null && signal.getSourceEntityId().equals(excludeSourceId)) {
                continue;
            }
            Object targetId = signal.getMetadata().get("targetEntityId");
            if (targetId instanceof UUID && targetId.equals(entityId)) {
                return true;
            }
        }
        return false;
    }

    public static List<AISignalData> getSignalsNear(BlockPosRef position, double maxDistance, long currentTick) {
        List<AISignalData> results = new ArrayList<>();

        for (List<AISignalData> signals : activeSignals.values()) {
            for (AISignalData signal : signals) {
                if (signal.isExpired(currentTick)) {
                    continue;
                }
                double intensity = signal.getIntensity(position);
                if (intensity > 0.0) {
                    results.add(signal);
                }
            }
        }

        results.sort((a, b) -> Double.compare(b.getIntensity(position), a.getIntensity(position)));
        return results;
    }

    public static void cleanupExpiredSignals(long currentTick) {
        for (List<AISignalData> signals : activeSignals.values()) {
            signals.removeIf(signal -> signal.isExpired(currentTick));
        }
        for (List<AISignalData> signals : entitySignals.values()) {
            signals.removeIf(signal -> signal.isExpired(currentTick));
        }
    }

    public static void clearSignalsForEntity(UUID entityId) {
        entitySignals.remove(entityId);
        for (List<AISignalData> signals : activeSignals.values()) {
            signals.removeIf(signal -> signal.getSourceEntityId().equals(entityId));
        }
    }

    public static void clearSignal(AISignalData signal) {
        if (signal == null) {
            return;
        }
        List<AISignalData> typeSignals = activeSignals.get(signal.getType());
        if (typeSignals != null) {
            typeSignals.remove(signal);
        }
        List<AISignalData> entitySignalsList = entitySignals.get(signal.getSourceEntityId());
        if (entitySignalsList != null) {
            entitySignalsList.remove(signal);
        }
    }
}

import com.biocraft.alien.layer_1_infrastructure.level_0.platform.ModLogger;
import com.biocraft.alien.layer_1_infrastructure.level_3.territory.TerritoryConfig;
import com.biocraft.alien.layer_2_systems.level_0.block.BlockRuleHelper;

import java.util.ArrayList;
import java.util.List;

/**
 * Space validation logic for hive expansion.
 */
public class ExpansionValidator {

    private static final int GROUND_LEVEL_OFFSET = 0;

    private ExpansionValidator() {
    }

    private static double calculateDistance2D(int x, int z) {
        return Math.sqrt(x * x + z * z);
    }

    public static ValidationResult canExpandToHive(
            WorldQueryPort query, WorldRef world, BlockPosRef centerPos, int targetRadius) {
        if (query == null || world == null || centerPos == null) {
            return ValidationResult.failure("Cannot validate without server world context");
        }
        int nestInnerRadius = TerritoryConfig.getNestInnerRadius();
        List<BlockPosRef> obstacles = scanExpansionArea(query, world, centerPos, nestInnerRadius, targetRadius);
        double obstaclePercentage = getObstaclePercentage(query, world, centerPos, targetRadius);

        double requiredClearance = TerritoryConfig.getExpansionClearanceRequired() / 100.0;
        if (obstaclePercentage > (1.0 - requiredClearance)) {
            return ValidationResult.failure(
                    String.format("Insufficient clearance: %.1f%% obstacles (need %.1f%% clear)",
                            obstaclePercentage * 100, requiredClearance * 100));
        }

        for (BlockPosRef obstacle : obstacles) {
            String blockId = query.blockIdAt(world, obstacle);
            if (isUnbreakableBlock(blockId)) {
                return ValidationResult.failure("Unbreakable obstacles detected: " + blockId);
            }
        }

        ModLogger.info("ExpansionValidator",
                String.format("✅ Expansion validation passed: %.1f%% clear space",
                        (1.0 - obstaclePercentage) * 100));

        return ValidationResult.success();
    }

    public static List<BlockPosRef> scanExpansionArea(
            WorldQueryPort query,
            WorldRef world,
            BlockPosRef centerPos,
            int innerRadius,
            int outerRadius) {
        List<BlockPosRef> obstacles = new ArrayList<>();

        for (int x = -outerRadius; x <= outerRadius; x++) {
            for (int z = -outerRadius; z <= outerRadius; z++) {
                if (Math.abs(x) <= innerRadius && Math.abs(z) <= innerRadius) {
                    continue;
                }

                double distance = calculateDistance2D(x, z);
                if (distance > innerRadius && distance <= outerRadius) {
                    BlockPosRef pos = centerPos.offset(x, GROUND_LEVEL_OFFSET, z);

                    if (hasObstacleAt(query, world, pos) || hasObstacleAt(query, world, pos.above())) {
                        obstacles.add(pos);
                    }
                }
            }
        }

        return obstacles;
    }

    public static double getObstaclePercentage(
            WorldQueryPort query, WorldRef world, BlockPosRef centerPos, int radius) {
        int totalPositions = 0;
        int obstaclePositions = 0;
        int nestInnerRadius = TerritoryConfig.getNestInnerRadius();

        for (int x = -radius; x <= radius; x++) {
            for (int z = -radius; z <= radius; z++) {
                if (Math.abs(x) <= nestInnerRadius && Math.abs(z) <= nestInnerRadius) {
                    continue;
                }

                double distance = calculateDistance2D(x, z);
                if (distance > nestInnerRadius && distance <= radius) {
                    totalPositions++;

                    BlockPosRef pos = centerPos.offset(x, GROUND_LEVEL_OFFSET, z);
                    if (hasObstacleAt(query, world, pos) || hasObstacleAt(query, world, pos.above())) {
                        obstaclePositions++;
                    }
                }
            }
        }

        return totalPositions > 0 ? (double) obstaclePositions / totalPositions : 0.0;
    }

    private static boolean hasObstacleAt(WorldQueryPort query, WorldRef world, BlockPosRef pos) {
        if (!query.isInBounds(world, pos)) {
            return true;
        }

        if (query.isAirAt(world, pos)) {
            return false;
        }

        String blockId = query.blockIdAt(world, pos);
        if (isUnbreakableBlock(blockId)) {
            return true;
        }

        return query.canOccludeAt(world, pos);
    }

    private static boolean isUnbreakableBlock(String blockId) {
        return BlockRuleHelper.isUnbreakable(blockId);
    }
}

        for (QueenTerritory territory : nearby) {
            UUID owner = territory.getQueenUUID();
            if (selfQueenId == null || owner == null || !owner.equals(selfQueenId)) {
                return true;
            }
        }
        return false;
    }

    public static boolean hasSuitableGround(WorldQueryPort query, WorldRef world, BlockPosRef pos) {
        if (query == null || world == null || pos == null) {
            return false;
        }
        if (!query.canOccludeAt(world, pos.below())) {
            return false;
        }
        if (!query.isAirAt(world, pos)) {
            return false;
        }
        return query.isAirAt(world, pos.above());
    }

    public static ValidationResult validateNestLocation(WorldQueryPort query, WorldRef world, BlockPosRef pos) {
        return validateNestLocation(query, world, null, pos);
    }

    public static ValidationResult validateNestLocation(
            WorldQueryPort query, WorldRef world, UUID claimingQueenId, BlockPosRef pos) {
        ValidationResult availability = canCreateNestAt(claimingQueenId, pos);
        if (!availability.isSuccess()) {
            return availability;
        }

        if (!hasSuitableGround(query, world, pos)) {
            return ValidationResult.failure("Ground is not suitable for nest building");
        }

        return ValidationResult.success();
    }
}

import java.util.Objects;
import java.util.Optional;
import java.util.UUID;

import com.biocraft.alien.layer_0_domain.level_0.lab.BreachStatus;
import com.biocraft.alien.layer_0_domain.level_0.lab.ContainmentIntegrity;

import com.biocraft.alien.layer_2_systems.level_1.containment.ContainmentOccupancyService;

/**
 * Containment authority that owns integrity/breach state and composes S3B occupancy.
 *
 * <p>Occupancy mutations always delegate to {@link ContainmentOccupancyService}.
 * Breach mutations update domain state only — no gameplay reactions.
 */
public final class ContainmentService {

    private ContainmentService() {
    }

    /**
     * Returns an uncached integrity status snapshot without mutating state.
     */
    public static ContainmentStatus query(ContainmentPort port) {
        Objects.requireNonNull(port, "port");
        return new ContainmentStatus(port.volumeId(), port.integrity(), port.controllerWorld());
    }

    /**
     * Query-only breach snapshot — no mutation, no gameplay.
     */
    public static BreachState queryBreach(ContainmentPort port) {
        Objects.requireNonNull(port, "port");
        BreachState state = port.breachState();
        return state == null ? BreachState.secure() : state;
    }

    /**
     * Query-only occupancy composition — delegates to S3B without writes.
     */
    public static ContainmentOccupancySnapshot queryOccupancy(ContainmentOccupancyPort occupancy) {
        return ContainmentOccupancyService.query(occupancy);
    }

    public static ValidationResult setIntegrity(ContainmentPort port, ContainmentIntegrity integrity) {
        Objects.requireNonNull(port, "port");
        if (integrity == null) {
            return ValidationResult.failure("containment integrity is required");
        }
        if (port.volumeId() == null) {
            return ValidationResult.failure("containment volume ID is required");
        }
        if (port.integrity() == integrity) {
            return ValidationResult.success();
        }
        port.setIntegrity(integrity);
        return ValidationResult.success();
    }

    /**
     * Sets integrity and clears occupancy when the result is BREACHED.
     */
    public static ValidationResult setIntegrity(
            ContainmentPort containment,
            ContainmentOccupancyPort occupancy,
            ContainmentIntegrity integrity) {
        Objects.requireNonNull(occupancy, "occupancy");
        ValidationResult result = setIntegrity(containment, integrity);
        if (result.isSuccess() && integrity == ContainmentIntegrity.BREACHED) {
            ContainmentOccupancyService.clearOnBreach(occupancy);
        }
        return result;
    }

    /**
     * Advances coarse integrity one step and syncs graded breach state.
     * Idempotent at BREACHED. Prefer {@link #damage} for graded failure.
     */
    public static ValidationResult degrade(ContainmentPort port, long nowMillis) {
        Objects.requireNonNull(port, "port");
        ContainmentIntegrity current = port.integrity();
        if (current == null) {
            return ValidationResult.failure("containment integrity is required");
        }
        return switch (current) {
            case INTACT -> {
                applyBreachState(port, new BreachState(BreachStatus.COMPROMISED, 50, Optional.empty()));
                yield ValidationResult.success();
            }
            case DEGRADED -> {
                applyBreachState(
                        port,
                        new BreachState(BreachStatus.BREACHED, 0, Optional.of(nowMillis)));
                yield ValidationResult.success();
            }
            case BREACHED -> ValidationResult.success();
        };
    }

    public static ValidationResult degrade(ContainmentPort port) {
        return degrade(port, 0L);
    }

    /**
     * Degrades integrity and clears occupancy once BREACHED.
     */
    public static ValidationResult degrade(
            ContainmentPort containment,
            ContainmentOccupancyPort occupancy,
            long nowMillis) {
        Objects.requireNonNull(occupancy, "occupancy");
        ValidationResult result = degrade(containment, nowMillis);
        if (result.isSuccess() && containment.integrity() == ContainmentIntegrity.BREACHED) {
            ContainmentOccupancyService.clearOnBreach(occupancy);
        }
        return result;
    }

    public static ValidationResult degrade(ContainmentPort containment, ContainmentOccupancyPort occupancy) {
        return degrade(containment, occupancy, 0L);
    }

    /**
     * Applies graded damage to breach state and syncs coarse integrity. No gameplay reactions.
     */
    public static ValidationResult damage(ContainmentPort port, int amount, long nowMillis) {
        Objects.requireNonNull(port, "port");
        if (amount < 0) {
            return ValidationResult.failure("damage amount cannot be negative");
        }
        if (port.volumeId() == null) {
            return ValidationResult.failure("containment volume ID is required");
        }
        BreachState current = queryBreach(port);
        BreachState next = current.damage(amount, nowMillis);
        if (next.equals(current)) {
            return ValidationResult.success();
        }
        applyBreachState(port, next);
        return ValidationResult.success();
    }

    /**
     * Damage that clears occupancy when breach state becomes failed.
     */
    public static ValidationResult damage(
            ContainmentPort containment,
            ContainmentOccupancyPort occupancy,
            int amount,
            long nowMillis) {
        Objects.requireNonNull(occupancy, "occupancy");
        ValidationResult result = damage(containment, amount, nowMillis);
        if (result.isSuccess() && queryBreach(containment).status().isFailed()) {
            ContainmentOccupancyService.clearOnBreach(occupancy);
        }
        return result;
    }

    /**
     * Repairs graded integrity when eligible. No gameplay reactions.
     */
    public static ValidationResult repair(ContainmentPort port, int amount) {
        Objects.requireNonNull(port, "port");
        if (amount < 0) {
            return ValidationResult.failure("repair amount cannot be negative");
        }
        if (port.volumeId() == null) {
            return ValidationResult.failure("containment volume ID is required");
        }
        BreachState current = queryBreach(port);
        if (!current.isRepairEligible()) {
            return ValidationResult.failure("containment is not eligible for repair");
        }
        BreachState next = current.repair(amount);
        if (next.equals(current)) {
            return ValidationResult.success();
        }
        applyBreachState(port, next);
        return ValidationResult.success();
    }

    public static ValidationResult disable(ContainmentPort port, long nowMillis) {
        Objects.requireNonNull(port, "port");
        BreachState next = queryBreach(port).disable(nowMillis);
        applyBreachState(port, next);
        return ValidationResult.success();
    }

    public static ValidationResult enable(ContainmentPort port) {
        Objects.requireNonNull(port, "port");
        BreachState next = queryBreach(port).enable();
        applyBreachState(port, next);
        return ValidationResult.success();
    }

    /**
     * Restores integrity and breach state to secure. Idempotent when already secure.
     */
    public static ValidationResult restore(ContainmentPort port) {
        Objects.requireNonNull(port, "port");
        applyBreachState(port, BreachState.secure());
        return ValidationResult.success();
    }

    /**
     * Admits an occupant only when integrity is secure; occupancy policy remains S3B.
     */
    public static ValidationResult admit(
            ContainmentPort containment,
            ContainmentOccupancyPort occupancy,
            LivingEntityRef occupant) {
        Objects.requireNonNull(containment, "containment");
        Objects.requireNonNull(occupancy, "occupancy");
        if (containment.integrity() == null || !containment.integrity().isSecure()) {
            return ValidationResult.failure("containment is not secure");
        }
        return ContainmentOccupancyService.admit(occupancy, occupant);
    }

    /**
     * Releases occupancy via S3B. Does not mutate integrity.
     */
    public static Optional<UUID> release(ContainmentOccupancyPort occupancy) {
        return ContainmentOccupancyService.release(occupancy);
    }

    /**
     * Clears occupancy via S3B. Does not mutate integrity.
     */
    public static void clearOnBreach(ContainmentOccupancyPort occupancy) {
        ContainmentOccupancyService.clearOnBreach(occupancy);
    }

    private static void applyBreachState(ContainmentPort port, BreachState next) {
        port.setBreachState(next);
        ContainmentIntegrity mapped = next.toIntegrity();
        if (port.integrity() != mapped) {
            port.setIntegrity(mapped);
        }
    }
}

/**
 * Command-based handling authority. RELEASE changes membership for existing entities only;
 * never creates biological entities or selects organism type.
 */
public final class HandlingService {

    private HandlingService() {
    }

    /**
     * Validates and delegates a handling request. Does not spawn entities.
     */
    public static HandlingResult request(
            HandlingRequest request,
            ContainmentPort containment,
            ContainmentOccupancyPort occupancy) {
        Objects.requireNonNull(request, "request");
        Objects.requireNonNull(containment, "containment");
        Objects.requireNonNull(occupancy, "occupancy");

        if (request.volumeId().isPresent()
                && !Objects.equals(request.volumeId().get(), containment.volumeId())) {
            return HandlingResult.rejected("handling volume does not match containment port");
        }

        return switch (request.action()) {
            case ADMIT -> admit(request, containment, occupancy);
            case RELEASE -> release(request, occupancy);
        };
    }

    private static HandlingResult admit(
            HandlingRequest request,
            ContainmentPort containment,
            ContainmentOccupancyPort occupancy) {
        if (containment.integrity() == null || !containment.integrity().isSecure()) {
            return HandlingResult.rejected("containment is not secure");
        }
        ValidationResult result = ContainmentOccupancyService.admit(occupancy, request.entity());
        if (!result.isSuccess()) {
            return HandlingResult.rejected(result.getReason());
        }
        return HandlingResult.ok();
    }

    private static HandlingResult release(HandlingRequest request, ContainmentOccupancyPort occupancy) {
        Optional<UUID> occupant = occupancy.occupantEntityId();
        if (occupant.isEmpty()) {
            return HandlingResult.rejected("containment has no occupant");
        }
        if (!Objects.equals(occupant.get(), request.entity().entityId())) {
            return HandlingResult.rejected("entity is not the current occupant");
        }
        ContainmentOccupancyService.release(occupancy);
        return HandlingResult.ok();
    }

    /**
     * Explicit surface documenting that handling never performs organism-type spawn.
     */
    public static boolean supportsOrganismSpawn(HandlingAction action) {
        Objects.requireNonNull(action, "action");
        return false;
    }
}

package com.biocraft.alien.layer_2_systems.level_2.parasite;

import com.biocraft.alien.layer_0_domain.level_0.ai.AISignalType;
import com.biocraft.alien.layer_0_domain.level_1.entity.LivingEntityRef;
import com.biocraft.alien.layer_2_systems.level_1.signals.AISignalSystem;

import java.util.Map;
import java.util.Objects;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;
import javax.annotation.Nullable;

/**
 * Single authority for transient lifecycle claims on a potential host.
 *
 * <p>Claims are distinct from infection/occupancy. Facehugger claims are represented by
 * AI signals; ovomorph hatch claims use a short-lived reservation because eggs do not emit
 * those signals.
 */
public final class HostClaimService {

    private static final long DEFAULT_OVOMORPH_CLAIM_TTL_MS = 60_000L;
    private static final Map<UUID, OvomorphClaim> OVOMORPH_CLAIMS_BY_HOST = new ConcurrentHashMap<>();

    private HostClaimService() {
        throw new UnsupportedOperationException("HostClaimService is a utility class");
    }

    public static boolean isHostClaimed(@Nullable LivingEntityRef host, long currentTick) {
        return host != null && isHostClaimed(host.entityId(), currentTick);
    }

    public static boolean isHostClaimed(@Nullable UUID hostId, long currentTick) {
        if (hostId == null) {
            return false;
        }
        return hasTargeted(hostId, currentTick)
                || hasReserved(hostId, currentTick)
                || hasAttached(hostId, currentTick)
                || hasOvomorphClaim(hostId);
    }

    public static boolean isHostClaimedByOther(
            @Nullable UUID hostId,
            @Nullable UUID claimantId,
            long currentTick) {
        if (hostId == null) {
            return false;
        }
        return AISignalSystem.hasSignalForEntityExcludingSource(
                        AISignalType.HOST_TARGETED, hostId, claimantId, currentTick)
                || AISignalSystem.hasSignalForEntityExcludingSource(
                        AISignalType.HOST_RESERVED, hostId, claimantId, currentTick)
                || AISignalSystem.hasSignalForEntityExcludingSource(
                        AISignalType.HOST_ATTACHED, hostId, claimantId, currentTick)
                || isOvomorphClaimedByOther(hostId, claimantId);
    }

    public static boolean hasTargeted(@Nullable UUID hostId, long currentTick) {
        return hostId != null
                && AISignalSystem.hasSignalForEntity(AISignalType.HOST_TARGETED, hostId, currentTick);
    }

    public static boolean hasReserved(@Nullable UUID hostId, long currentTick) {
        return hostId != null
                && AISignalSystem.hasSignalForEntity(AISignalType.HOST_RESERVED, hostId, currentTick);
    }

    public static boolean hasAttached(@Nullable UUID hostId, long currentTick) {
        return hostId != null
                && AISignalSystem.hasSignalForEntity(AISignalType.HOST_ATTACHED, hostId, currentTick);
    }

    public static void claimForOvomorph(@Nullable UUID hostId, @Nullable UUID eggId) {
        claimForOvomorph(hostId, eggId, DEFAULT_OVOMORPH_CLAIM_TTL_MS);
    }

    public static void claimForOvomorph(@Nullable UUID hostId, @Nullable UUID eggId, long ttlMs) {
        if (hostId == null || eggId == null) {
            return;
        }
        purgeExpiredOvomorphClaims();
        OvomorphClaim existing = OVOMORPH_CLAIMS_BY_HOST.get(hostId);
        if (existing != null && !existing.claimantId().equals(eggId) && !existing.isExpired()) {
            return;
        }
        long ttl = Math.max(1_000L, ttlMs);
        OVOMORPH_CLAIMS_BY_HOST.put(
                hostId,
                new OvomorphClaim(eggId, System.currentTimeMillis() + ttl));
    }

    public static void releaseOvomorphClaims(@Nullable UUID eggId) {
        if (eggId == null) {
            return;
        }
        Iterator<Map.Entry<UUID, OvomorphClaim>> iterator =
                OVOMORPH_CLAIMS_BY_HOST.entrySet().iterator();
        while (iterator.hasNext()) {
            Map.Entry<UUID, OvomorphClaim> entry = iterator.next();
            if (Objects.equals(entry.getValue().claimantId(), eggId)) {
                iterator.remove();
            }
        }
    }

    public static void releaseOvomorphClaim(@Nullable UUID hostId, @Nullable UUID eggId) {
        if (hostId == null) {
            return;
        }
        OvomorphClaim claim = OVOMORPH_CLAIMS_BY_HOST.get(hostId);
        if (claim != null && (eggId == null || claim.claimantId().equals(eggId))) {
            OVOMORPH_CLAIMS_BY_HOST.remove(hostId, claim);
        }
    }

    private static boolean hasOvomorphClaim(UUID hostId) {
        purgeExpiredOvomorphClaims();
        return OVOMORPH_CLAIMS_BY_HOST.containsKey(hostId);
    }

    private static boolean isOvomorphClaimedByOther(UUID hostId, @Nullable UUID claimantId) {
        purgeExpiredOvomorphClaims();
        OvomorphClaim claim = OVOMORPH_CLAIMS_BY_HOST.get(hostId);
        return claim != null
                && (claimantId == null || !claim.claimantId().equals(claimantId));
    }

    private static void purgeExpiredOvomorphClaims() {
        OVOMORPH_CLAIMS_BY_HOST.entrySet().removeIf(entry -> entry.getValue().isExpired());
    }

    private record OvomorphClaim(UUID claimantId, long expiresAtMs) {
        private boolean isExpired() {
            return System.currentTimeMillis() >= expiresAtMs;
        }
    }
}

        // Production path: chestburster → drone. Growth/praetor are shelved (FEAT-GROWTH-01 / FEAT-PRAETOR-01).
        Map<String, TransformationHandler> chestbursterTransforms = new HashMap<>();
        chestbursterTransforms.put("drone", entity -> transformTo(entity, XenomorphType.DRONE));
        REGISTRY.put(XenomorphType.CHESTBURSTER, chestbursterTransforms);

        Map<String, TransformationHandler> droneTransforms = new HashMap<>();
        droneTransforms.put("queen", entity -> transformTo(entity, XenomorphType.QUEEN));
        REGISTRY.put(XenomorphType.DRONE, droneTransforms);
    }

    private static boolean transformTo(LivingEntityRef entity, XenomorphType targetType) {
        if (!BoundaryServices.isInitialized()) {
            return false;
        }
        if (BoundaryServices.entityQuery().isClientSide(entity)) {
            return false;
        }
        return BoundaryServices.entitySpawn().transformXenomorph(entity, targetType) != null;
    }

    public static boolean transformEntity(LivingEntityRef entity, String targetStage) {
        if (EntityValidationHelper.isInvalid(entity, BoundaryServices.entityQuery()) || targetStage == null || !BoundaryServices.isInitialized()) {
            return false;
        }

        XenomorphType sourceType = BoundaryServices.formQuery().resolveForm(entity);
        if (sourceType == null) {
            ModLogger.warn("TransformationSystem", "No transformation handler for unknown entity");
            return false;
        }

        Map<String, TransformationHandler> entityTransforms = REGISTRY.get(sourceType);
        if (entityTransforms == null) {
            ModLogger.warn("TransformationSystem", "No transformation handler found for type: " + sourceType);
            return false;
        }

        TransformationHandler handler = entityTransforms.get(targetStage.toLowerCase());
        if (handler == null) {
            ModLogger.warn("TransformationSystem", "Unknown transformation target for " + sourceType + ": " + targetStage);
            return false;
        }

        return handler.transform(entity);
    }

    public static boolean canTransform(LivingEntityRef entity, String targetStage) {
        if (EntityValidationHelper.isInvalid(entity, BoundaryServices.entityQuery()) || targetStage == null || !BoundaryServices.isInitialized()) {
            return false;
        }

        XenomorphType sourceType = BoundaryServices.formQuery().resolveForm(entity);
        if (sourceType == null) {
            return false;
        }

        Map<String, TransformationHandler> entityTransforms = REGISTRY.get(sourceType);
        return entityTransforms != null && entityTransforms.containsKey(targetStage.toLowerCase());
    }

    public static String getTransformationRequirements(LivingEntityRef entity, String targetStage) {
        if (targetStage == null || entity == null || !BoundaryServices.isInitialized()) {
            return "Invalid entity or target stage";
        }

        if (EntityValidationHelper.isInvalid(entity, BoundaryServices.entityQuery())) {
            return "Invalid entity or target stage";
        }

        if (!canTransform(entity, targetStage)) {
            XenomorphType sourceType = BoundaryServices.isInitialized()
                    ? BoundaryServices.formQuery().resolveForm(entity)
                    : null;
            return "Transformation not available for " + sourceType + " to " + targetStage;
        }

        EntityConfig config = GrowthEntityTypeHelper.getGrowthConfig(entity);
        if (config != null) {
            XenomorphType sourceType = BoundaryServices.formQuery().resolveForm(entity);
            return String.format("%s must be fully matured (%d ticks) to transform to %s",
                    sourceType, config.maturationTime, targetStage);
        }

        XenomorphType sourceType = BoundaryServices.formQuery().resolveForm(entity);
        return String.format("%s must meet maturation requirements to transform to %s", sourceType, targetStage);
    }
}

/**
 * Admission Transaction orchestrator.
 *
 * <p>Coordinates Specimen Source → containment validation → Realization → Handling ADMIT →
 * settle. Does not branch on carrier families (spawn egg / bucket / Net) — that logic belongs
 * in {@link SpecimenSource} or Realization.
 *
 * <p>TODO(FEATURE-ADMIT-001): foundation orchestrator.
 */
public final class AdmissionTransaction {

    private AdmissionTransaction() {
    }

    /**
     * Run one admission transaction to commit or rollback.
     */
    public static AdmissionResult run(
            SpecimenSource source,
            SpecimenRealizationPort realization,
            ContainmentPort containment,
            ContainmentOccupancyPort occupancy,
            AdmissionContext context) {
        Objects.requireNonNull(source, "source");
        Objects.requireNonNull(realization, "realization");
        Objects.requireNonNull(containment, "containment");
        Objects.requireNonNull(occupancy, "occupancy");
        Objects.requireNonNull(context, "context");

        if (!occupancy.isStructureFormed()) {
            return AdmissionResult.rollback("containment structure is not formed");
        }
        if (!Objects.equals(context.volumeId(), containment.volumeId())) {
            return AdmissionResult.rollback("admission volume does not match containment port");
        }
        if (containment.integrity() == null || !containment.integrity().isSecure()) {
            return AdmissionResult.rollback("containment is not secure");
        }
        if (occupancy.occupantEntityId().isPresent()) {
            return AdmissionResult.rollback("containment is occupied");
        }
        if (!source.canExtract()) {
            return AdmissionResult.rollback("specimen source cannot extract");
        }

        Optional<PortableSpecimen> extracted = source.extract();
        if (extracted.isEmpty()) {
            return AdmissionResult.rollback("specimen source extract failed");
        }
        PortableSpecimen portable = extracted.get();

        InteriorVolume interior = context.interior();
        if (!ContainmentFitRules.fits(
                interior.extentX(),
                interior.extentY(),
                interior.extentZ(),
                portable.extents().width(),
                portable.extents().height())) {
            return AdmissionResult.rollback(fitFailureReason(interior, portable.extents()));
        }

        Optional<LivingEntityRef> realized = realization.realize(portable, context);
        if (realized.isEmpty()) {
            return AdmissionResult.rollback("specimen realization failed");
        }
        LivingEntityRef entity = realized.get();

        HandlingResult admit = HandlingService.request(
                HandlingRequest.admit(entity, context.volumeId()), containment, occupancy);
        if (!admit.accepted()) {
            realization.discard(entity);
            return AdmissionResult.rollback(
                    admit.reason() == null ? "handling admit rejected" : admit.reason());
        }

        SpecimenSettlement settlement = source.settleAfterAdmission();
        if (!settlement.applied()) {
            // Occupancy stays; settlement failure does not roll back membership.
            return AdmissionResult.commit();
        }
        return AdmissionResult.commit();
    }

    private static String fitFailureReason(InteriorVolume interior, SpecimenExtents extents) {
        int needHorizontal = (int) Math.ceil(extents.width() - 1.0e-9);
        int needHeight = (int) Math.ceil(extents.height() - 1.0e-9);
        return ContainmentFitRules.REJECT_DOES_NOT_FIT
                + String.format(
                        Locale.ROOT,
                        " (specimen %.1f×%.1f needs interior ≥ %d wide / %d tall; pen interior %d×%d×%d)",
                        extents.width(),
                        extents.height(),
                        needHorizontal,
                        needHeight,
                        interior.sizeX(),
                        interior.sizeY(),
                        interior.sizeZ());
    }
}

/**
 * Orchestrates entity growth lifecycle for xenomorphs (process, transform, stage queries).
 *
 * <p><b>Timer ownership (TECH-GROWTH-001):</b> {@link MaturationSystem} owns maturation timer
 * CRUD. This class delegates {@link #getMaturationTimer}, {@link #setMaturationTimer}, and
 * {@link #reduceMaturationTimer} to {@code MaturationSystem} so L3 callers can use one growth
 * facade without importing maturation internals. That dual surface is intentional — do not
 * duplicate timer state here.
 *
 * FILE STATUS: FLUX
 */
public class EntityGrowthSystem {

    public enum GrowthStage {
        CHESTBURSTER,
        GROWTH,
        DRONE,
        PRAETOR,
        QUEEN
    }

    public static boolean processGrowth(@Nonnull LivingEntityRef entity) {
        if (EntityValidationHelper.isInvalid(entity, BoundaryServices.entityQuery())) {
            return false;
        }

        MaturationSystem.processMaturation(entity);

        if (canTransform(entity)) {
            String targetStage = getNextStage(entity);
            if (targetStage != null) {
                return transformEntity(entity, targetStage);
            }
        }

        return false;
    }

    public static boolean canTransform(@Nonnull LivingEntityRef entity) {
        if (EntityValidationHelper.isInvalid(entity, BoundaryServices.entityQuery())) {
            return false;
        }

        EntityConfig config = GrowthEntityTypeHelper.getGrowthConfig(entity);
        if (config == null) {
            return false;
        }

        // maturation_time <= 0 means "never auto-mature" (JSON omit defaults to 0).
        if (!config.canMature || config.maturationTime <= 0) {
            return false;
        }

        int timer = MaturationSystem.getMaturationTimer(entity);
        return timer >= config.maturationTime;
    }

    public static boolean transformEntity(@Nonnull LivingEntityRef entity, String targetStage) {
        if (EntityValidationHelper.isInvalid(entity, BoundaryServices.entityQuery()) || targetStage == null || !BoundaryServices.isInitialized()) {
            return false;
        }

        ModLogger.info("EntityGrowthSystem", "Transforming "
                + BoundaryServices.presentation().displayName(entity)
                + " to stage: " + targetStage);

        return TransformationSystem.transformEntity(entity, targetStage);
    }

    public static GrowthStage getGrowthStage(LivingEntityRef entity) {
        if (EntityValidationHelper.isInvalid(entity, BoundaryServices.entityQuery())) {
            return null;
        }

        XenomorphType xenomorphType = XenomorphTypeHelper.getXenomorphType(entity);
        if (xenomorphType == null) {
            return null;
        }

        return switch (xenomorphType) {
            case CHESTBURSTER -> GrowthStage.CHESTBURSTER;
            case DRONE -> GrowthStage.DRONE;
            default -> null;
        };
    }

    public static double getGrowthProgress(@Nonnull LivingEntityRef entity) {
        if (EntityValidationHelper.isInvalid(entity, BoundaryServices.entityQuery())) {
            return 0.0;
        }

        EntityConfig config = GrowthEntityTypeHelper.getGrowthConfig(entity);
        if (config == null || config.maturationTime <= 0) {
            return 0.0;
        }

        int timer = MaturationSystem.getMaturationTimer(entity);
        return Math.min(1.0, (double) timer / config.maturationTime);
    }

    public static String getNextStage(@Nonnull LivingEntityRef entity) {
        if (EntityValidationHelper.isInvalid(entity, BoundaryServices.entityQuery())) {
            return null;
        }

        EntityConfig config = GrowthEntityTypeHelper.getGrowthConfig(entity);
        if (config == null || config.matureTo == null || config.matureTo.isEmpty()) {
            return null;
        }

        return config.matureTo;
    }

    public static int getMaturationTimer(@Nonnull LivingEntityRef entity) {
        return MaturationSystem.getMaturationTimer(entity);
    }

    public static void setMaturationTimer(@Nonnull LivingEntityRef entity, int timer) {
        MaturationSystem.setMaturationTimer(entity, timer);
    }

    public static void reduceMaturationTimer(@Nonnull LivingEntityRef entity, int reduction) {
        MaturationSystem.reduceMaturationTimer(entity, reduction);
    }

    public static boolean canGrow(LivingEntityRef entity) {
        if (EntityValidationHelper.isInvalid(entity, BoundaryServices.entityQuery())) {
            return false;
        }
        return GrowthEntityTypeHelper.canMature(entity);
    }
}

/**

 *
 * <p>Latched host identity is owned by {@link com.biocraft.alien.layer_0_domain.level_2.parasite.AttachmentBinding}
 * via {@link AttachmentStatePort#getTarget}/{@link AttachmentStatePort#setTarget}. Never use combat
 * {@code Mob.getTarget()} to identify an attached host.
 */
public class FacehuggerAttachmentManager {

    private static final String LOG = "FacehuggerAttachment";

    private FacehuggerAttachmentManager() {
    }

    // TODO(BUG-LIFECYCLE-002): Stage 5 polish — tall/awkward host latch edges (human path reliable).
    public static boolean attemptAttachment(@Nonnull LivingEntityRef facehugger, @Nonnull LivingEntityRef host) {
        if (!supportsAttachment(facehugger) || EntityValidationHelper.isInvalid(host, BoundaryServices.entityQuery()) || !BoundaryServices.isInitialized()) {
            return false;
        }

        if (isSpent(facehugger)) {
            return false;
        }

        if (BoundaryServices.entityQuery().countAttachedFacehuggers(host) > 0) {
            ModLogger.info(LOG, "Host " + BoundaryServices.presentation().displayName(host)
                    + " already has an attached facehugger - denying attachment");
            return false;
        }

        if (isEntityBeingFacehugged(host)) {
            ModLogger.info(LOG, "Host " + BoundaryServices.presentation().displayName(host)
                    + " is already being facehugged");
            return false;
        }

        if (GestationManager.isGestating(host)) {
            ModLogger.info(LOG, "Host " + BoundaryServices.presentation().displayName(host)
                    + " is already gestating - denying attachment");
            return false;
        }

        if (BoundaryServices.presentation().distance(facehugger, host) > AttachmentConfigHelper.getAttachmentDistance()) {
            return false;
        }

        startAttachment(facehugger, host);
        return true;
    }

    private static void startAttachment(@Nonnull LivingEntityRef facehugger, @Nonnull LivingEntityRef host) {
        ModLogger.info(LOG, "Facehugger " + BoundaryServices.presentation().displayName(facehugger)
                + " starting attachment to " + BoundaryServices.presentation().displayName(host)
                + " at distance " + String.format("%.2f", BoundaryServices.presentation().distance(facehugger, host)));

        setTarget(facehugger, host);
        setTicks(facehugger, 0);
        // ATTACHING is synched; remaining latch feel is BUG-LIFECYCLE-001 (pose hold + face standoff).
        setState(facehugger, AttachmentState.ATTACHING);
        BoundaryServices.presentation().positionOnHost(facehugger, host, AttachmentConfigHelper.getFaceHeightRatio());

        BoundaryServices.presentation().setInvulnerable(facehugger, true);
        float latchYaw = getHostYaw(host) + (float) AttachmentConfigHelper.getRotationOffset();
        BoundaryServices.presentation().setRotation(facehugger, latchYaw, 0.0f);
        applyAttachmentEffects(facehugger, host);

        ModLogger.info(LOG, "Facehugger attached to " + BoundaryServices.presentation().displayName(host)
                + " - immobile and invulnerable");
    }

    public static boolean handleAttachmentTick(@Nonnull LivingEntityRef facehugger, @Nonnull LivingEntityRef host,
                                               int attachmentTicks) {
        if (EntityValidationHelper.isInvalid(host, BoundaryServices.entityQuery())) {
            ModLogger.info(LOG, "Attachment tick " + attachmentTicks + ": Host is dead or null - ending attachment");
            BoundaryServices.presentation().applyMagicDamage(facehugger, Float.MAX_VALUE);
            return false;
        }

        // Reposition first while latched so host movement cannot trip max-distance abort.
        BoundaryServices.presentation().positionOnHost(facehugger, host, AttachmentConfigHelper.getFaceHeightRatio());

        AttachmentState latchedState = getState(facehugger);
        boolean latched = latchedState == AttachmentState.ATTACHING || latchedState == AttachmentState.ATTACHED;
        if (!latched
                && BoundaryServices.presentation().distance(facehugger, host)
                        > AttachmentConfigHelper.getAttachmentDistance()) {
            stopAttachment(facehugger, host);
            return false;
        }

        int attachmentDuration = AttachmentConfigHelper.getAttachmentDuration();
        int implantationDelay = AttachmentConfigHelper.getImplantationDelay();
        int totalRequired = attachmentDuration + implantationDelay;

        if (attachmentTicks >= attachmentDuration) {
            if (attachmentTicks >= totalRequired) {
                ModLogger.info(LOG, "Implantation delay completed - proceeding with implantation");
                completeImplantation(facehugger, host);
                return false;
            }
            int remainingDelay = totalRequired - attachmentTicks;
            if (attachmentTicks % 40 == 0) {
                ModLogger.info(LOG, "Facehugger " + BoundaryServices.presentation().displayName(facehugger)
                        + " holding on to " + BoundaryServices.presentation().displayName(host)
                        + " - " + remainingDelay + " ticks until implantation");
            }
        }

        return true;
    }

    private static void completeImplantation(@Nonnull LivingEntityRef facehugger, @Nonnull LivingEntityRef host) {
        ModLogger.info(LOG, "Facehugger " + BoundaryServices.presentation().displayName(facehugger)
                + " completed implantation on " + BoundaryServices.presentation().displayName(host));
        // TODO(FEATURE-DNA-002): After DNA gestation fields exist, spawn/implant outcome from host DNA id.
        GestationManager.applyGestationEffect(host);
        ModLogger.info(LOG, "Gestation effect applied to " + BoundaryServices.presentation().displayName(host));
        detachFacehugger(facehugger, host);
    }

    private static void detachFacehugger(@Nonnull LivingEntityRef facehugger, @Nonnull LivingEntityRef host) {
        ModLogger.info(LOG, "Starting facehugger detachment process");
        // Clear DNA/no-AI debuffs even if MobEffect.onEffectRemoved is skipped.
        HostEffectManager.removeHostEffects(host);
        removeAttachmentEffect(host);
        removeImplantationEffect(facehugger);

        BoundaryServices.presentation().setNoGravity(facehugger, false);
        BoundaryServices.presentation().setInvulnerable(facehugger, false);
        // Drop off the host face so spent wander does not leave the entity floating mid-air.
        BoundaryServices.presentation().setDeltaMovement(facehugger, 0, -0.2, 0);
        clearBinding(facehugger);

        FacehuggerSpentTransitionServices.spentTransition().initializeSpentSession(facehugger);

        ModLogger.info(LOG, "Facehugger " + BoundaryServices.presentation().displayName(facehugger)
                + " marked as spent - will wander before dying");
    }

    /**
     * Clears latch presentation / effects when attachment aborts without implant
     * (goal cancel, distance abort). Safe no-op when boundary services are down.
     */
    public static void abortLatchCleanup(@Nonnull LivingEntityRef facehugger, @Nullable LivingEntityRef host) {
        if (!BoundaryServices.isInitialized()) {
            return;
        }
        stopAttachment(facehugger, host);
    }

    /**
     * Cleanup contract: resolve host from binding first, remove effects, restore gravity, clear binding.
     * Never gate host effect removal on combat {@code getTarget()}.
     */
    private static void stopAttachment(@Nonnull LivingEntityRef facehugger, @Nullable LivingEntityRef host) {
        LivingEntityRef resolved = resolveBoundHost(facehugger, host);
        if (resolved != null) {
            HostEffectManager.removeHostEffects(resolved);
            removeAttachmentEffect(resolved);
        }
        removeImplantationEffect(facehugger);
        BoundaryServices.presentation().setNoGravity(facehugger, false);
        BoundaryServices.presentation().setInvulnerable(facehugger, false);
        clearBinding(facehugger);
    }

    @Nullable
    private static LivingEntityRef resolveBoundHost(
            @Nonnull LivingEntityRef facehugger, @Nullable LivingEntityRef host) {
        if (host != null) {
            return host;
        }
        if (!supportsAttachment(facehugger)) {
            return null;
        }
        return port().getTarget(facehugger);
    }

    private static void clearBinding(@Nonnull LivingEntityRef facehugger) {
        if (supportsAttachment(facehugger)) {
            port().setTarget(facehugger, null);
        }
    }

    public static boolean isEntityBeingFacehugged(@Nonnull LivingEntityRef entity) {
        return entity != null && BoundaryServices.isInitialized()
                && BoundaryServices.effects().hasEffect(entity, DomainEffectKey.FACEHUGGER_ATTACHMENT);
    }

    public static int getAttachmentDuration() {
        return AttachmentConfigHelper.getAttachmentDuration();
    }

    public static double getAttachmentDistance() {
        return AttachmentConfigHelper.getAttachmentDistance();
    }

    public static boolean handleAttachmentStateBusinessLogic(@Nonnull LivingEntityRef facehugger,
                                                               @Nonnull LivingEntityRef host) {
        if (!supportsAttachment(facehugger) || EntityValidationHelper.isInvalid(host, BoundaryServices.entityQuery())) {
            return false;
        }

        return switch (getState(facehugger)) {
            case READY_TO_ATTACH -> {
                ModLogger.info(LOG, "Facehugger " + BoundaryServices.presentation().displayName(facehugger)
                        + " starting attachment business logic to " + BoundaryServices.presentation().displayName(host));
                if (attemptAttachment(facehugger, host)) {
                    setState(facehugger, AttachmentState.ATTACHING);
                    ModLogger.info(LOG, "Facehugger " + BoundaryServices.presentation().displayName(facehugger)
                            + " successfully started attachment to " + BoundaryServices.presentation().displayName(host));
                    yield true;
                }
                ModLogger.warn(LOG, "Facehugger " + BoundaryServices.presentation().displayName(facehugger)
                        + " failed to start attachment to " + BoundaryServices.presentation().displayName(host));
                setState(facehugger, AttachmentState.FREE);
                yield false;
            }
            case ATTACHING, ATTACHED, DETACHING, SPENT_WANDERING -> true;
            default -> true;
        };
    }

    private static void applyAttachmentEffects(@Nonnull LivingEntityRef facehugger, @Nonnull LivingEntityRef host) {
        int fullDuration = AttachmentConfigHelper.getAttachmentDuration()
                + AttachmentConfigHelper.getImplantationDelay();
        EffectApplyOptions options = new EffectApplyOptions(false, false, true);
        BoundaryServices.effects().apply(host, DomainEffectKey.FACEHUGGER_ATTACHMENT, fullDuration, 0, options);
        BoundaryServices.effects().apply(facehugger, DomainEffectKey.FACEHUGGER_IMPLANTATION, fullDuration, 0,
                new EffectApplyOptions(false, false, false));
    }

    private static void removeAttachmentEffect(LivingEntityRef host) {
        BoundaryServices.effects().remove(host, DomainEffectKey.FACEHUGGER_ATTACHMENT);
    }

    private static void removeImplantationEffect(LivingEntityRef facehugger) {
        BoundaryServices.effects().remove(facehugger, DomainEffectKey.FACEHUGGER_IMPLANTATION);
    }

    private static boolean supportsAttachment(LivingEntityRef facehugger) {
        return AttachmentServices.isInitialized() && AttachmentServices.attachment().supports(facehugger);
    }

    private static AttachmentStatePort port() {
        return AttachmentServices.attachment();
    }

    private static AttachmentState getState(LivingEntityRef facehugger) {
        return port().getState(facehugger);
    }

    private static void setState(@Nonnull LivingEntityRef facehugger, @Nonnull AttachmentState state) {
        port().setState(facehugger, state);
    }

    private static void setTicks(@Nonnull LivingEntityRef facehugger, int ticks) {
        port().setTicks(facehugger, ticks);
    }

    private static void setTarget(@Nonnull LivingEntityRef facehugger, @Nullable LivingEntityRef host) {
        port().setTarget(facehugger, host);
    }

    private static boolean isSpent(LivingEntityRef facehugger) {
        return getState(facehugger) == AttachmentState.SPENT_WANDERING;
    }

    private static float getHostYaw(LivingEntityRef host) {

    }

import com.biocraft.alien.layer_0_domain.level_0.entity.EntityKind;
import com.biocraft.alien.layer_0_domain.level_1.entity.LivingEntityRef;

import java.util.function.Predicate;

/**
 * Targeting policy for xenomorph combat goals.
 * Combat is broader than facehugger host suitability (e.g. slimes are killable).
 */
public final class XenomorphTargetingHelper {

    private XenomorphTargetingHelper() {
        throw new UnsupportedOperationException("XenomorphTargetingHelper is a utility class");
    }

    public static Predicate<LivingEntityRef> isValidTarget() {
        return XenomorphTargetingHelper::isValidTargetEntity;
    }

    public static boolean isValidTargetEntity(LivingEntityRef entity) {
        if (!BoundaryServices.isInitialized()) {
            return false;
        }
        EntityQueryPort query = BoundaryServices.entityQuery();
        if (entity == null || !query.isValid(entity)) {
            return false;
        }
        if (query.classify(entity) == EntityKind.MOD_XENOMORPH) {
            return false;
        }
        EntityKind kind = query.classify(entity);
        if (kind == EntityKind.PLAYER && query.isCreativePlayer(entity)) {
            return false;
        }
        // Protect claimed and occupied hosts so combat does not interrupt the lifecycle.
        if (isLifecycleProtected(entity, BoundaryServices.presentation().currentGameTime(entity))) {
            return false;
        }
        // Players / villagers / animals remain valid combat prey without host-registry lookup.
        if (kind == EntityKind.PLAYER || kind == EntityKind.VILLAGER || kind == EntityKind.ANIMAL) {
            return true;
        }
        String mobName = query.encodeId(entity).map(XenomorphTargetingHelper::registryPath).orElse(null);
        if (mobName == null) {
            return false;
        }
        // Combat: any mob listed in hosts.json, including unsuitable (slime, etc.).
        // Facehugger/ovomorph keep MobHostRegistry.isSuitableForXenomorph separately.
        return MobHostRegistry.getHostType(mobName) != null;
    }

    /**
     * Claimed and occupied are distinct lifecycle facts, but both exclude proactive combat.
     */
    public static boolean isLifecycleProtected(LivingEntityRef entity, long currentTick) {
        return entity != null
                && (HostAvailabilityService.isHostOccupied(entity)
                || HostClaimService.isHostClaimed(entity, currentTick));
    }

    public static int getTargetPriority(LivingEntityRef entity) {
        if (!BoundaryServices.isInitialized() || entity == null) {
            return 7;
        }
        return switch (BoundaryServices.entityQuery().classify(entity)) {
            case PLAYER -> 2;
            case VILLAGER -> 3;
            case ANIMAL -> 4;
            default -> 7;
        };
    }

    public record ScoredTarget(LivingEntityRef ref, double distanceSq) {
    }

    public static void addScoredCandidate(
            java.util.List<ScoredTarget> candidates,
            LivingEntityRef ref,
            double distanceSq) {
        if (isValidTargetEntity(ref)) {
            candidates.add(new ScoredTarget(ref, distanceSq));
        }
    }

    /**
     * Selects the highest-priority target; ties break on nearest distance.
     */
    public static LivingEntityRef selectBestTarget(java.util.List<ScoredTarget> candidates) {
        LivingEntityRef best = null;
        int bestPriority = Integer.MAX_VALUE;
        double bestDistanceSq = Double.MAX_VALUE;

        for (ScoredTarget candidate : candidates) {
            if (candidate == null || candidate.ref() == null) {
                continue;
            }
            int priority = getTargetPriority(candidate.ref());
            double distanceSq = candidate.distanceSq();
            if (priority < bestPriority || (priority == bestPriority && distanceSq < bestDistanceSq)) {
                best = candidate.ref();
                bestPriority = priority;
                bestDistanceSq = distanceSq;
            }
        }
        return best;
    }

    private static String registryPath(String encodeId) {
        if (encodeId == null || encodeId.isBlank()) {
            return null;
        }
        int separator = encodeId.indexOf(':');
        return separator >= 0 ? encodeId.substring(separator + 1) : encodeId;
    }
}

import javax.annotation.Nonnull;
import javax.annotation.Nullable;

import com.biocraft.alien.layer_0_domain.level_0.parasite.AttachmentState;
import com.biocraft.alien.layer_0_domain.level_1.entity.LivingEntityRef;
import com.biocraft.alien.layer_0_domain.level_1.parasite.AttachmentRules;
import com.biocraft.alien.layer_0_domain.level_2.parasite.AttachmentStatePort;
import com.biocraft.alien.layer_1_infrastructure.level_2.entity.AttachmentConfigHelper;
import com.biocraft.alien.layer_2_systems.level_0.entity.LivingEntityChecks;
import com.biocraft.alien.layer_2_systems.level_0.parasite.FacehuggerAttachmentEntity;
import com.biocraft.alien.layer_2_systems.level_0.services.AttachmentServices;
import com.biocraft.alien.layer_2_systems.level_0.services.BoundaryServices;
import com.biocraft.alien.layer_2_systems.level_3.parasite.FacehuggerAttachmentManager;
import com.biocraft.alien.layer_2_systems.level_5.parasite.ParasiteLifecycleBootstrap;

/**
 * Orchestrates facehugger attachment ticks, spent transition, and goal-cancel resets.
 *
 * <p>S2B playtest-confirmed (2026-07-24): attach → latch → implant → spent stroll.
 */
public final class FacehuggerTickWorkflow {

    public static final class SpentCallbacks {
        private final Runnable die;
        private final Runnable discard;

        public SpentCallbacks(@Nonnull Runnable die, @Nonnull Runnable discard) {
            this.die = die;
            this.discard = discard;
        }
    }

    private FacehuggerTickWorkflow() {
    }

    /**
     * Domain field reset when implantation completes and the facehugger becomes spent.
     * Caller applies MC side effects (goal rebuild, logging).
     */
    public static void applyMarkAsSpent(@Nonnull FacehuggerAttachmentEntity facehugger) {
        facehugger.setSpentTimer(0);
        facehugger.setAttachmentTarget(null);
        facehugger.setAttachmentTicks(0);
        facehugger.setAttachmentState(AttachmentState.SPENT_WANDERING);
        facehugger.setAttachmentRotationSet(false);
        facehugger.setPendingTargetUUID(null);
    }

    /**
     * Domain field reset when the host dies during attachment.
     * Caller applies MC death side effects.
     */
    public static void applyHostDeathReset(@Nonnull FacehuggerAttachmentEntity facehugger) {
        facehugger.setAttachmentTarget(null);
        facehugger.setAttachmentTicks(0);
        facehugger.setAttachmentState(AttachmentState.FREE);
    }

    /**
     * Domain field reset when the attachment goal is cancelled / stopped.
     * Preserves spent / detaching terminal states — goal stop also fires after successful implant.
     */
    public static void applyAttachmentGoalCancelled(@Nonnull FacehuggerAttachmentEntity facehugger) {
        AttachmentState state = facehugger.getAttachmentState();
        if (AttachmentRules.isSpentState(state) || state == AttachmentState.DETACHING) {
            return;
        }
        facehugger.setAttachmentTarget(null);
        facehugger.setAttachmentTicks(0);
        facehugger.setAttachmentState(AttachmentRules.onAttachmentGoalCancelled());
    }

    public static int processSpentState(@Nonnull LivingEntityRef facehugger, int spentTimer,
                                        @Nonnull SpentCallbacks callbacks) {
        if (!AttachmentRules.isSpentState(getAttachmentState(facehugger))) {
            return -1;
        }

        int nextTimer = spentTimer + 1;

        int spentWanderTime = AttachmentConfigHelper.getSpentWanderTime();
        int deathAnimationTicks = AttachmentConfigHelper.getDeathAnimationTime();

        // Die once when wander ends; discard only after death animation so vanilla death plays.
        boolean crossedWanderEnd = AttachmentRules.shouldDieAfterSpentWander(nextTimer, spentWanderTime)
                && !AttachmentRules.shouldDieAfterSpentWander(spentTimer, spentWanderTime);
        if (crossedWanderEnd) {
            callbacks.die.run();
        }
        boolean crossedDeathEnd = AttachmentRules.shouldDiscardAfterDeathAnimation(
                nextTimer, spentWanderTime, deathAnimationTicks)
                && !AttachmentRules.shouldDiscardAfterDeathAnimation(
                        spentTimer, spentWanderTime, deathAnimationTicks);
        if (crossedDeathEnd) {
            callbacks.discard.run();
        }
        return nextTimer;
    }

    public static void processAttachmentTick(@Nonnull LivingEntityRef facehugger, @Nonnull LivingEntityRef target) {
        if (LivingEntityChecks.isInvalid(target, BoundaryServices.entityQuery())) {
            return;
        }

        AttachmentState attachmentState = getAttachmentState(facehugger);

        if (attachmentState == AttachmentState.READY_TO_ATTACH) {
            FacehuggerAttachmentManager.handleAttachmentStateBusinessLogic(facehugger, target);
        }

        attachmentState = getAttachmentState(facehugger);
        if (!AttachmentRules.shouldProcessImplantationTick(attachmentState)) {
            return;
        }

        boolean shouldContinue = ParasiteLifecycleBootstrap.processAttachmentTick(
                facehugger, target, getAttachmentTicks(facehugger));

        if (shouldContinue) {
            setAttachmentTicks(facehugger, getAttachmentTicks(facehugger) + 1);
        } else if (!AttachmentRules.isSpentState(getAttachmentState(facehugger))
                && !AttachmentRules.shouldProcessImplantationTick(getAttachmentState(facehugger))) {
            // Successful implant sets SPENT before returning false. Mid-latch abort must not
            // FREE-reset while still ATTACHING/ATTACHED (goal-cancel race wiped implant).
            resetAttachmentState(facehugger);
        }
    }

    public static boolean shouldDieWithHost(@Nonnull AttachmentState state, boolean hostAlive) {
        return AttachmentRules.shouldFacehuggerDieWithHost(state, hostAlive);
    }

    private static AttachmentState getAttachmentState(@Nonnull LivingEntityRef facehugger) {
        AttachmentStatePort port = activePort(facehugger);
        return port != null ? port.getState(facehugger) : AttachmentState.FREE;
    }

    private static int getAttachmentTicks(@Nonnull LivingEntityRef facehugger) {
        AttachmentStatePort port = activePort(facehugger);
        return port != null ? port.getTicks(facehugger) : 0;
    }

    private static void setAttachmentTicks(@Nonnull LivingEntityRef facehugger, int ticks) {
        AttachmentStatePort port = activePort(facehugger);
        if (port != null) {
            port.setTicks(facehugger, ticks);
        }
    }

    private static void resetAttachmentState(@Nonnull LivingEntityRef facehugger) {
        AttachmentStatePort port = activePort(facehugger);
        if (port != null) {
            port.setTarget(facehugger, null);
            port.setTicks(facehugger, 0);
            port.setState(facehugger, AttachmentState.FREE);
        }
    }

    @Nullable
    private static AttachmentStatePort activePort(@Nonnull LivingEntityRef facehugger) {
        if (!AttachmentServices.isInitialized()) {
            return null;
        }
        AttachmentStatePort port = AttachmentServices.attachment();
        return port.supports(facehugger) ? port : null;
    }
