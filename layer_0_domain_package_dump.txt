    public AdmissionResult {
        if (!committed) {
            reason = Objects.requireNonNull(reason, "rollback reason required");
        }
    }

    public static AdmissionResult commit() {
        return new AdmissionResult(true, null);

        return new AdmissionResult(false, reason);
    }

    public boolean rolledBack() {
        return !committed;

package com.biocraft.alien.layer_0_domain.level_0.ai;

import java.util.List;

/**
 * Pure preferred-food matching (entity type key vs configured food names).
 * Uses exact or whole-token match — never substring (avoids {@code pig} → {@code piglin}).
 */
public final class PreferredFoodRules {

    private PreferredFoodRules() {}

    public static boolean matches(String entityTypeKey, List<String> preferredFood) {
        if (entityTypeKey == null || preferredFood == null || preferredFood.isEmpty()) {
            return false;
        }
        String key = entityTypeKey.toLowerCase();
        String path = key.contains(":") ? key.substring(key.indexOf(':') + 1) : key;
        return preferredFood.stream().anyMatch(food -> matchesOne(key, path, food));
    }

    private static boolean matchesOne(String fullKey, String path, String food) {
        if (food == null || food.isEmpty()) {
            return false;
        }
        String f = food.toLowerCase();
        if (fullKey.equals(f) || path.equals(f)) {
            return true;
        }
        for (String token : path.split("_")) {
            if (token.equals(f)) {
                return true;
            }
        }
        return false;
    }
}

/**
 * Domain rules for survival behavior decisions.
 *
 * Contains pure survival logic shared across implementations.
 * Does not depend on Minecraft, NeoForge, configuration loading,
 * or infrastructure.
 *
 * FILE STATUS: NEW
 *
 * @author BioCraft
 * @since 0.1.0
 */
public final class SurvivalBehaviorRules {

    private SurvivalBehaviorRules() {
        throw new UnsupportedOperationException(
                "SurvivalBehaviorRules is a utility class and cannot be instantiated");
    }


    /**
     * Determines whether suffocation should trigger escape behavior.
     *
     * @param airRatio current air ratio (0.0 - 1.0)
     * @param currentlyStuck whether the entity is stuck
     * @param criticalThreshold immediate escape threshold
     * @param lowThreshold stuck escape threshold
     *
     * @return true if escape should occur
     */
    public static boolean shouldEscapeSuffocation(
            double airRatio,
            boolean currentlyStuck,
            double criticalThreshold,
            double lowThreshold) {

        if (airRatio <= criticalThreshold) {
            return true;
        }

        return currentlyStuck && airRatio <= lowThreshold;
    }


    /**
     * Determines whether an escape attempt cooldown has expired.
     *
     * @param cooldownTicks remaining cooldown
     *
     * @return true when an escape attempt is allowed
     */
    public static boolean canAttemptEscape(int cooldownTicks) {
        return cooldownTicks <= 0;
    }


    /**
     * Determines whether an entity has exceeded its stuck threshold.
     *
     * @param stuckTicks current stuck duration
     * @param threshold configured threshold
     *
     * @return true when entity is stuck
     */
    public static boolean isStuck(
            int stuckTicks,
            int threshold) {

        return stuckTicks >= threshold;
    }
}

import java.util.Objects;
import java.util.Set;

/**
 * Domain view of block break eligibility lists.
 * Populated from infrastructure config; consumed by {@link BlockBreakPolicy}.
 *
 * FILE STATUS: STABLE
 *
 * @author BioCraft
 * @since 0.1.0
 */
public final class BlockBreakRules {

    private final Set<String> unbreakable;
    private final Set<String> breakable;

    /**
     * @param unbreakable block registry IDs that must never be broken
     * @param breakable   block registry IDs on the explicit whitelist
     */
    public BlockBreakRules(Set<String> unbreakable, Set<String> breakable) {
        this.unbreakable = Collections.unmodifiableSet(
                Objects.requireNonNullElse(unbreakable, Set.of()));
        this.breakable = Collections.unmodifiableSet(
                Objects.requireNonNullElse(breakable, Set.of()));
    }

    /**
     * @param blockId registry ID
     * @return true if listed as unbreakable
     */
    public boolean isUnbreakable(String blockId) {
        if (blockId == null || blockId.isBlank()) {
            return false;
        }
        return unbreakable.contains(blockId);
    }

    /**
     * @param blockId registry ID
     * @return true if listed on the breakable whitelist
     */
    public boolean isExplicitlyBreakable(String blockId) {
        if (blockId == null || blockId.isBlank()) {
            return false;
        }
        return breakable.contains(blockId);
    }

    public Set<String> unbreakable() {
        return unbreakable;
    }

    public Set<String> breakable() {
        return breakable;
    }
}

package com.biocraft.alien.layer_0_domain.level_0.entity;

import javax.annotation.Nonnull;

/**
 * Pure server movement-form state machine. Interrupted transitions cancel and
 * re-enter a stable phase (no stuck TransitionToCrawl / TransitionToStand).
 */
public final class MovementFormFsm {

    public static final double MOVE_THRESHOLD_SQR = 1.0E-6D;

    private MovementFormFsm() {
    }

    public static boolean isMovingHorizontally(double horizontalDistanceSqr) {
        return horizontalDistanceSqr > MOVE_THRESHOLD_SQR;
    }

    /**
     * Advances phase/timer for one tick. Mutates {@code state} in place.
     *
     * @param state  mutable phase + remaining ticks + crawlingForm flag
     * @param moving whether horizontal locomotion intent is active this tick
     */
    public static void tick(MovementFormState state, boolean moving) {
        switch (state.phase) {
            case STANDING -> {
                if (moving) {
                    state.phase = MovementFormPhase.TRANSITION_TO_CRAWL;
                    state.ticksRemaining = MovementFormTiming.CRAWL_DELAY_TICKS;
                }
            }
            case TRANSITION_TO_CRAWL -> {
                if (!moving) {
                    // Interrupt: cancel crawl transition; still standing hitbox.
                    state.phase = MovementFormPhase.STANDING;
                    state.ticksRemaining = 0;
                    state.crawlingForm = false;
                } else if (--state.ticksRemaining <= 0) {
                    state.phase = MovementFormPhase.CRAWLING;
                    state.ticksRemaining = 0;
                    state.crawlingForm = true;
                }
            }
            case CRAWLING -> {
                if (!moving) {
                    state.phase = MovementFormPhase.TRANSITION_TO_STAND;
                    state.ticksRemaining = MovementFormTiming.STAND_DELAY_TICKS;
                }
            }
            case TRANSITION_TO_STAND -> {
                if (moving) {
                    // Interrupt: cancel stand transition; still crawling hitbox.
                    state.phase = MovementFormPhase.CRAWLING;
                    state.ticksRemaining = 0;
                    state.crawlingForm = true;
                } else if (--state.ticksRemaining <= 0) {
                    state.phase = MovementFormPhase.STANDING;
                    state.ticksRemaining = 0;
                    state.crawlingForm = false;
                }
            }
        }
    }

    /** Mutable FSM snapshot applied by entity hosts each server tick. */
    public static final class MovementFormState {
        @Nonnull
        public MovementFormPhase phase = MovementFormPhase.STANDING;
        public int ticksRemaining;
        public boolean crawlingForm;

        public void copyFrom(@Nonnull MovementFormPhase phase, int ticksRemaining, boolean crawlingForm) {
            this.phase = phase;
            this.ticksRemaining = ticksRemaining;
            this.crawlingForm = crawlingForm;
        }
    }
}

import java.util.Objects;
import javax.annotation.Nonnull;

/**
 * Server-owned stand/crawl gameplay form phase (TECH-ANIM-TRANSITION-001).
 * Hitbox form flips only when entering {@link #CRAWLING} or {@link #STANDING}.
 */
public enum MovementFormPhase {
    STANDING,
    TRANSITION_TO_CRAWL,
    CRAWLING,
    TRANSITION_TO_STAND;

    @Nonnull
    public static MovementFormPhase fromOrdinal(int ordinal) {
        MovementFormPhase[] values = values();
        if (ordinal < 0 || ordinal >= values.length) {
            return STANDING;
        }
        return Objects.requireNonNull(values[ordinal]);
    }

    /** True while hitbox / crawl idle should use crawling dimensions. */
    public boolean usesCrawlingHitbox() {
        return this == CRAWLING || this == TRANSITION_TO_STAND;
    }
}

package com.biocraft.alien.layer_0_domain.level_0.lab;

/**
 * Pure containment capacity fit policy.
 *
 * <p>Capacity is determined by interior block volume extents, never by shell model thickness
 * or presentation shapes. Default pen-tier interior is one block cell ({@link #DEFAULT_PEN_INTERIOR_EXTENT}).
 */
public final class ContainmentFitRules {

    public static final String REJECT_DOES_NOT_FIT = "entity does not fit containment interior";

    /**
     * Standard containment-pen interior extents (blocks) for the hollow 3×3×3 tier —
     * single-cell interior at controller+(0,1,0).
     */
    public static final double DEFAULT_PEN_INTERIOR_EXTENT = 1.0;

    private ContainmentFitRules() {
    }

    /**
     * Whether an entity with the given bounding extents fits the interior volume.
     *
     * @param interiorX interior width (blocks)
     * @param interiorY interior height (blocks)
     * @param interiorZ interior depth (blocks)
     * @param entityWidth entity XZ diameter (blocks), e.g. Minecraft {@code bbWidth}
     * @param entityHeight entity height (blocks), e.g. Minecraft {@code bbHeight}
     */
    public static boolean fits(
            double interiorX,
            double interiorY,
            double interiorZ,
            double entityWidth,
            double entityHeight) {
        if (interiorX <= 0 || interiorY <= 0 || interiorZ <= 0) {
            return false;
        }
        if (entityWidth <= 0 || entityHeight <= 0) {
            return false;
        }
        return entityWidth <= interiorX
                && entityWidth <= interiorZ
                && entityHeight <= interiorY;
    }

    /**
     * Fit check against the default containment-pen interior extents for this tier.
     */
    public static boolean fitsContainmentPenInterior(double entityWidth, double entityHeight) {
        return fits(
                DEFAULT_PEN_INTERIOR_EXTENT,
                DEFAULT_PEN_INTERIOR_EXTENT,
                DEFAULT_PEN_INTERIOR_EXTENT,
                entityWidth,
                entityHeight);
    }
}

package com.biocraft.alien.layer_0_domain.level_0.nest;

/**
 * Pure relative block offsets for nest-structure scan patterns (no Minecraft types).
 */
public final class NestStructureScanOffsets {

    private NestStructureScanOffsets() {
    }

    /**
     * Horizontal offsets in the foundation plane ({@code y = 0} relative to nest center).
     */
    public static int[][] foundationRingOffsets(int scanRadius) {
        return squareRing(scanRadius, true);
    }

    /**
     * Perimeter offsets used for wall detection ({@code y = 1} relative), edges of the square only.
     */
    public static int[][] wallPerimeterOffsets(int scanRadius) {
        return squareRing(scanRadius, false);
    }

    /**
     * Full horizontal square offsets for decoration / nearby-block probes ({@code y} supplied by caller).
     */
    public static int[][] horizontalSquareOffsets(int scanRadius) {
        return squareRing(scanRadius, true);
    }

    private static int[][] squareRing(int radius, boolean includeInterior) {
        int count = 0;
        for (int x = -radius; x <= radius; x++) {
            for (int z = -radius; z <= radius; z++) {
                boolean edge = x == -radius || x == radius || z == -radius || z == radius;
                if (includeInterior || edge) {
                    count++;
                }
            }
        }
        int[][] offsets = new int[count][2];
        int i = 0;
        for (int x = -radius; x <= radius; x++) {
            for (int z = -radius; z <= radius; z++) {
                boolean edge = x == -radius || x == radius || z == -radius || z == radius;
                if (includeInterior || edge) {
                    offsets[i][0] = x;
                    offsets[i][1] = z;
                    i++;
                }
            }
        }
        return offsets;
    }
}

import java.util.Locale;
import java.util.Objects;
import java.util.Set;

/**
 * Governance allowlist / deny-list for research fact keys.
 *
 * <p>Facts represent captured observations, not derived conclusions.
 */
public final class ResearchFactKeys {

    public static final String MOVEMENT = "movement";
    public static final String CONTAINMENT_STATUS = "containment_status";
    public static final String SAMPLE_KIND = "sample_kind";
    public static final String OBSERVED_TICK = "observed_tick";

    private static final Set<String> ALLOWED = Set.of(
            MOVEMENT,
            CONTAINMENT_STATUS,
            SAMPLE_KIND,
            OBSERVED_TICK);

    private static final Set<String> FORBIDDEN = Set.of(
            "threat",
            "danger",
            "aggression",
            "evolution_stage");

    private ResearchFactKeys() {
    }

    public static boolean isAllowed(String key) {
        if (key == null) {
            return false;
        }
        String normalized = normalize(key);
        return ALLOWED.contains(normalized) && !FORBIDDEN.contains(normalized);
    }

    public static boolean isForbidden(String key) {
        if (key == null) {
            return false;
        }
        return FORBIDDEN.contains(normalize(key));
    }

    public static void requireAllowed(String key) {
        Objects.requireNonNull(key, "fact key required");
        String normalized = normalize(key);
        if (FORBIDDEN.contains(normalized)) {
            throw new IllegalArgumentException("forbidden interpretation fact key: " + key);
        }
        if (!ALLOWED.contains(normalized)) {
            throw new IllegalArgumentException("fact key not on capture allowlist: " + key);
        }
    }

    public static Set<String> allowedKeys() {
        return ALLOWED;
    }

    private static String normalize(String key) {
        return key.trim().toLowerCase(Locale.ROOT);
    }
}

package com.biocraft.alien.layer_0_domain.level_0.spatial;

/**
 * Domain 3D vector for position search algorithms without Minecraft types.
 */
public record Vec3Ref(double x, double y, double z) {

    public Vec3Ref add(Vec3Ref other) {
        return new Vec3Ref(x + other.x, y + other.y, z + other.z);
    }

    public Vec3Ref subtract(Vec3Ref other) {
        return new Vec3Ref(x - other.x, y - other.y, z - other.z);
    }

    public Vec3Ref scale(double factor) {
        return new Vec3Ref(x * factor, y * factor, z * factor);
    }

    public Vec3Ref normalize() {
        double length = Math.sqrt(x * x + y * y + z * z);
        if (length < 1.0E-4D) {
            return this;
        }
        return new Vec3Ref(x / length, y / length, z / length);
    }

    public Vec3Ref yRot(float angle) {
        double cos = Math.cos(angle);
        double sin = Math.sin(angle);
        return new Vec3Ref(x * cos + z * sin, y, z * cos - x * sin);
    }
}

public final class AIContextPolicy {

    private AIContextPolicy() {
    }

    public static AIContext fromAttachmentActive(boolean attached) {
        return attached ? AIContext.ATTACHMENT : AIContext.IDLE;
    }

    public record ContextThresholds(
            double fightingDistanceSq,
            double chasingDistanceSq,
            double stalkingDistanceSq
    ) {
    }

    public record ContextInputs(
            boolean hasTarget,
            boolean targetAlive,
            boolean crouching,
            double distanceToTargetSq,
            boolean recentlyHurt,
            boolean pathfinding,
            Double distanceToPathTargetSq
    ) {
    }

    public record TransitionSpeedModifiers(
            float normalModifier,
            float stalkingModifier,
            float chasingModifier,
            float fightingModifier
    ) {
    }

    public static AIContext classify(ContextInputs inputs, ContextThresholds thresholds) {
        if (thresholds == null) {
            throw new IllegalArgumentException("ContextThresholds is required");
        }
        if (inputs.hasTarget() && inputs.targetAlive()) {
            if (inputs.distanceToTargetSq() <= thresholds.fightingDistanceSq()) {
                return AIContext.FIGHTING;
            }
            if (inputs.distanceToTargetSq() <= thresholds.chasingDistanceSq()) {
                return AIContext.CHASING;
            }
        }
        if (inputs.crouching()) {
            return AIContext.STALKING;
        }
        if (inputs.recentlyHurt()) {
            return AIContext.COMBAT;
        }
        return AIContext.NORMAL;
    }

    public static float transitionSpeedModifier(AIContext context, TransitionSpeedModifiers modifiers) {
        if (modifiers == null) {
            throw new IllegalArgumentException("TransitionSpeedModifiers is required");
        }
        return switch (context) {
            case STALKING -> modifiers.stalkingModifier();
            case CHASING -> modifiers.chasingModifier();
            case FIGHTING, COMBAT -> modifiers.fightingModifier();
            default -> modifiers.normalModifier();
        };
    }
}

package com.biocraft.alien.layer_0_domain.level_1.blocks;

import com.biocraft.alien.layer_0_domain.level_0.blocks.BlockBreakRules;

/**
 * Domain policy for block breaking eligibility.
 *
 * Rules:
 * - Entity must be allowed to break blocks ({@code canBreakBlocks})
 * - Unbreakable blocks are never broken
 * - Unrestricted scope may break any non-unbreakable block
 * - Otherwise only explicitly whitelisted blocks may be broken
 *
 * Capability flags and rules lists are supplied by callers (systems/infra mapping).
 * This class has no infrastructure dependencies.
 *
 * FILE STATUS: STABLE
 *
 * @author BioCraft
 * @since 0.1.0
 */
public final class BlockBreakPolicy {

    private BlockBreakPolicy() {
        throw new UnsupportedOperationException(
                "BlockBreakPolicy is a utility class and cannot be instantiated");
    }

    /**
     * Determine if an entity may break a given block.
     *
     * @param blockId             registry ID (e.g., "minecraft:glass")
     * @param canBreakBlocks      whether the entity type may break blocks at all
     * @param unrestrictedScope   true to allow any non-unbreakable block
     * @param rules               domain break rules (unbreakable / whitelist)
     * @return true if this entity may break this block
     */
    public static boolean mayBreakBlock(
            String blockId,
            boolean canBreakBlocks,
            boolean unrestrictedScope,
            BlockBreakRules rules) {

        if (blockId == null || blockId.isBlank() || rules == null || !canBreakBlocks) {
            return false;
        }

        if (rules.isUnbreakable(blockId)) {
            return false;
        }

        if (unrestrictedScope) {
            return true;
        }

        return rules.isExplicitlyBreakable(blockId);
    }
}

package com.biocraft.alien.layer_0_domain.level_1.lab;

import java.util.Objects;
import java.util.Optional;

import com.biocraft.alien.layer_0_domain.level_0.lab.BreachStatus;
import com.biocraft.alien.layer_0_domain.level_0.lab.ContainmentIntegrity;

/**
 * Immutable graded breach state. Mutations return new instances; no gameplay side effects.
 *
 * <p>{@code integrityLevel} is 0–100. Status is derived from level except when {@link BreachStatus#DISABLED}.
 */
public record BreachState(
        BreachStatus status,
        int integrityLevel,
        Optional<Long> breachTimestampMillis
) {

    public static final int MAX_INTEGRITY = 100;
    public static final int MIN_INTEGRITY = 0;

    public BreachState {
        status = Objects.requireNonNull(status, "breach status required");
        if (integrityLevel < MIN_INTEGRITY || integrityLevel > MAX_INTEGRITY) {
            throw new IllegalArgumentException("integrityLevel must be 0–100");
        }
        breachTimestampMillis = Objects.requireNonNull(
                breachTimestampMillis, "breach timestamp optional required");
    }

    public static BreachState secure() {
        return new BreachState(BreachStatus.SECURE, MAX_INTEGRITY, Optional.empty());
    }

    public static BreachStatus statusForLevel(int integrityLevel) {
        if (integrityLevel < MIN_INTEGRITY || integrityLevel > MAX_INTEGRITY) {
            throw new IllegalArgumentException("integrityLevel must be 0–100");
        }
        if (integrityLevel == MAX_INTEGRITY) {
            return BreachStatus.SECURE;
        }
        if (integrityLevel >= 70) {
            return BreachStatus.WARNING;
        }
        if (integrityLevel >= 1) {
            return BreachStatus.COMPROMISED;
        }
        return BreachStatus.BREACHED;
    }

    /**
     * Applies damage without gameplay reactions. Idempotent at floor 0 / DISABLED.
     */
    public BreachState damage(int amount, long nowMillis) {
        if (amount < 0) {
            throw new IllegalArgumentException("damage amount cannot be negative");
        }
        if (status == BreachStatus.DISABLED || amount == 0) {
            return this;
        }
        int nextLevel = Math.max(MIN_INTEGRITY, integrityLevel - amount);
        BreachStatus nextStatus = statusForLevel(nextLevel);
        Optional<Long> timestamp = breachTimestampMillis;
        if (nextStatus == BreachStatus.BREACHED && breachTimestampMillis.isEmpty()) {
            timestamp = Optional.of(nowMillis);
        }
        return new BreachState(nextStatus, nextLevel, timestamp);
    }

    /**
     * Repairs integrity when eligible. Clears breach timestamp once no longer failed.
     */
    public BreachState repair(int amount) {
        if (amount < 0) {
            throw new IllegalArgumentException("repair amount cannot be negative");
        }
        if (!isRepairEligible() || amount == 0) {
            return this;
        }
        int nextLevel = Math.min(MAX_INTEGRITY, integrityLevel + amount);
        BreachStatus nextStatus = statusForLevel(nextLevel);
        Optional<Long> timestamp = nextStatus.isFailed() ? breachTimestampMillis : Optional.empty();
        return new BreachState(nextStatus, nextLevel, timestamp);
    }

    public BreachState disable(long nowMillis) {
        if (status == BreachStatus.DISABLED) {
            return this;
        }
        Optional<Long> timestamp = breachTimestampMillis.isPresent()
                ? breachTimestampMillis
                : Optional.of(nowMillis);
        return new BreachState(BreachStatus.DISABLED, integrityLevel, timestamp);
    }

    public BreachState enable() {
        if (status != BreachStatus.DISABLED) {
            return this;
        }
        BreachStatus nextStatus = statusForLevel(integrityLevel);
        Optional<Long> timestamp = nextStatus.isFailed() ? breachTimestampMillis : Optional.empty();
        return new BreachState(nextStatus, integrityLevel, timestamp);
    }

    public boolean isRepairEligible() {
        return status != BreachStatus.DISABLED && integrityLevel < MAX_INTEGRITY;
    }

    /**
     * Coarse integrity view for ContainedIntegrity / admission policy.
     */
    public ContainmentIntegrity toIntegrity() {
        return switch (status) {
            case SECURE -> ContainmentIntegrity.INTACT;
            case WARNING, COMPROMISED -> ContainmentIntegrity.DEGRADED;
            case BREACHED, DISABLED -> ContainmentIntegrity.BREACHED;
        };
    }
}

    private AttachmentConditionRules() {

    public static List<AttachmentCondition> dedupePreserveOrder(Collection<AttachmentCondition> conditions) {
        if (conditions == null || conditions.isEmpty()) {
            return List.of();
        }
        Set<AttachmentCondition> seen = EnumSet.noneOf(AttachmentCondition.class);
        List<AttachmentCondition> ordered = new ArrayList<>();
        for (AttachmentCondition condition : conditions) {
            if (condition != null && seen.add(condition)) {
                ordered.add(condition);
            }
        }
        return Collections.unmodifiableList(ordered);
    }

    public static boolean includes(Collection<AttachmentCondition> conditions, AttachmentCondition needle) {
        return conditions != null && needle != null && conditions.contains(needle);
    }
}

package com.biocraft.alien.layer_0_domain.level_1.parasite;

import com.biocraft.alien.layer_0_domain.level_0.spatial.Vec3Ref;

/**
 * Pure geometry for parasite attachment seat on a host AABB (not bone-specific).
 *
 * <p>TODO(BUG-LIFECYCLE-002): Stage 5 polish — tall/awkward host latch edges.
 * TODO(BUG-HOST-001): Stage 5 polish — Enderman seat distance (recognition OK).
 */
public final class AttachmentGeometry {

    private AttachmentGeometry() {
        throw new UnsupportedOperationException("AttachmentGeometry is a utility class");
    }

    /**
     * Forward distance from host origin to the latch seat along look yaw.
     * Half-width reaches the AABB face; head depth uses head size (width + crown above eyes),
     * not full body height — tall hosts like Enderman stay close to the face.
     *
     * @param eyeHeightFromFeet standing eye height relative to feet; {@code <= 0} falls back to height ratio proxies
     */
    public static double faceStandoffForward(double hostWidth, double hostHeight, double eyeHeightFromFeet) {
        double halfExtent = Math.max(0.0, hostWidth) * 0.5;
        double crownAboveEye = eyeHeightFromFeet > 0.05
                ? Math.max(0.0, hostHeight - eyeHeightFromFeet)
                : Math.max(0.2, hostHeight * 0.12);
        double headDepth = Math.max(0.15, Math.min(hostWidth * 0.55, crownAboveEye * 1.25 + 0.08));
        return halfExtent + headDepth;
    }

    /**
     * Vertical offset from host feet to the face seat. Prefers eye height when available.
     */
    public static double faceHeightOffset(double hostHeight, double heightRatio, double eyeHeightFromFeet) {
        if (eyeHeightFromFeet > 0.05) {
            return eyeHeightFromFeet;
        }
        return hostHeight * heightRatio;
    }

    /**
     * Attachment point on the host face region: eye/face height, nudged forward by look yaw.
     */
    public static Vec3Ref attachmentPoint(
            double hostX,
            double hostY,
            double hostZ,
            float hostYawDegrees,
            double hostWidth,
            double hostHeight,
            double heightRatio,
            double eyeHeightFromFeet) {
        double forward = faceStandoffForward(hostWidth, hostHeight, eyeHeightFromFeet);
        float yawRad = hostYawDegrees * ((float) Math.PI / 180.0f);
        double x = hostX - Math.sin(yawRad) * forward;
        double z = hostZ + Math.cos(yawRad) * forward;
        double y = hostY + faceHeightOffset(hostHeight, heightRatio, eyeHeightFromFeet);
        return new Vec3Ref(x, y, z);
    }

    public static double distance(Vec3Ref a, Vec3Ref b) {
        if (a == null || b == null) {
            return Double.MAX_VALUE;
        }
        double dx = a.x() - b.x();
        double dy = a.y() - b.y();
        double dz = a.z() - b.z();
        return Math.sqrt(dx * dx + dy * dy + dz * dz);
    }

    /** Horizontal reach is authoritative for beginning an attachment entry. */
    public static double horizontalDistance(Vec3Ref a, Vec3Ref b) {
        if (a == null || b == null) {
            return Double.MAX_VALUE;
        }
        double dx = a.x() - b.x();
        double dz = a.z() - b.z();
        return Math.sqrt(dx * dx + dz * dz);
    }

    /** True when geometry permits presentation to begin the attachment entry. */
    public static boolean canBeginAttachment(double horizontalDistance, double latchReach) {
        return horizontalDistance <= latchReach;
    }
}

/**
 * Pure domain rules for facehugger attachment FSM and invariants.
 * Tunable values belong in {@code layer_1_infrastructure.level_4.entity.AttachmentConfigHelper}.
 *
 * FILE STATUS: STABLE
 */
public final class AttachmentRules {

    private AttachmentRules() {
        throw new UnsupportedOperationException("AttachmentRules is a utility class and cannot be instantiated");
    }

    public static boolean isSpentState(AttachmentState state) {
        return state == AttachmentState.SPENT_WANDERING;
    }

    public static AttachmentState nextApproachState(AttachmentState current, double distance, double minDistance) {
        if (current == AttachmentState.SPENT_WANDERING || current == AttachmentState.DETACHING) {
            return current;
        }
        // Once latched, stay latched — distance jitter must not demote to APPROACHING
        // (that pauses implantation ticks and matches the "attaches then falls off" playtest bug).
        if (current == AttachmentState.ATTACHING || current == AttachmentState.ATTACHED) {
            return current;
        }
        if (distance > minDistance) {
            return AttachmentState.APPROACHING;
        }
        if (current == AttachmentState.FREE || current == AttachmentState.APPROACHING) {
            return AttachmentState.READY_TO_ATTACH;
        }
        return current;
    }

    public static boolean shouldRedirectDamageToHost(AttachmentState state, int attachmentTicks) {
        return attachmentTicks > 0
                && (state == AttachmentState.READY_TO_ATTACH
                || state == AttachmentState.ATTACHING
                || state == AttachmentState.ATTACHED);
    }

    public static boolean shouldFacehuggerDealDamage() {
        return false;

    public static boolean isImmovableDuringAttachment(AttachmentState state, int attachmentTicks) {
        return state == AttachmentState.READY_TO_ATTACH
                || state == AttachmentState.ATTACHING
                || state == AttachmentState.ATTACHED
                || (state == AttachmentState.DETACHING && attachmentTicks > 0);
    }

    /** True when spent wander time is over and the facehugger should begin dying. */
    public static boolean shouldDieAfterSpentWander(int nextTimer, int spentWanderTime) {
        return nextTimer >= spentWanderTime;

    @Deprecated

            int nextTimer, int spentWanderTime, int deathAnimationTicks) {
        if (deathAnimationTicks <= 0) {
            return shouldDieAfterSpentWander(nextTimer, spentWanderTime);
        }
        return nextTimer >= spentWanderTime + deathAnimationTicks;
    }

    public static boolean shouldProcessImplantationTick(AttachmentState state) {
        return state == AttachmentState.ATTACHING || state == AttachmentState.ATTACHED;
    }

    public static boolean shouldFacehuggerDieWithHost(AttachmentState state, boolean hostAlive) {
        if (hostAlive) {
            return false;
        }
        return state == AttachmentState.READY_TO_ATTACH
                || state == AttachmentState.ATTACHING
                || state == AttachmentState.ATTACHED;
    }

    /**
     * Tick-window check only. AI runs before implantation ticks, so at
     * {@code attachmentTicks == totalDuration} the goal must also keep running while
     * {@link #shouldProcessImplantationTick(AttachmentState)} is true — otherwise the
     * goal cancels and resets to FREE before implant can fire.
     */
    public static boolean isAttachmentInProgress(int attachmentTicks, int totalDuration) {
        return attachmentTicks < totalDuration;
    }

    public static AttachmentState onAttachmentGoalCancelled() {
        return AttachmentState.FREE;
    }
}

package com.biocraft.alien.layer_0_domain.level_1.spatial;

import com.biocraft.alien.layer_0_domain.level_0.spatial.Vec3Ref;

/**
 * Domain policy for spatial avoidance strength and flee direction.
 *
 * FILE STATUS: STABLE
 */
public final class AvoidancePolicy {

    private AvoidancePolicy() {
    }

    /**
     * Avoidance strength from {@code 0.0} (at or beyond min distance) to {@code 1.0} (coincident).
     */
    public static double computeAvoidanceStrength(double distance, int minDistance) {
        if (minDistance <= 0 || distance >= minDistance) {
            return 0.0;
        }
        return 1.0 - (distance / minDistance);
    }

    public static boolean shouldFlee(double avoidanceStrength) {
        return avoidanceStrength > 0.0;
    }

    public static boolean shouldLogStrongAvoidance(double avoidanceStrength) {
        return avoidanceStrength > 0.5;
    }

    /**
     * Unit vector away from {@code other}, scaled by {@code avoidanceStrength}.
     */
    public static Vec3Ref fleeDirection(Vec3Ref self, Vec3Ref other, double avoidanceStrength) {
        if (avoidanceStrength <= 0.0 || self == null || other == null) {
            return new Vec3Ref(0.0, 0.0, 0.0);
        }
        return self.subtract(other).normalize().scale(avoidanceStrength);
    }
}

package com.biocraft.alien.layer_0_domain.level_2.lab;

import com.biocraft.alien.layer_0_domain.level_0.lab.ContainmentFitRules;
import com.biocraft.alien.layer_0_domain.level_0.lab.HorizontalFacing;
import com.biocraft.alien.layer_0_domain.level_0.lab.InteriorVolume;
import com.biocraft.alien.layer_0_domain.level_1.lab.ShellCell;
import com.biocraft.alien.layer_0_domain.level_0.lab.ShellRole;
import com.biocraft.alien.layer_0_domain.level_0.spatial.BlockPosRef;
import com.biocraft.alien.layer_0_domain.level_0.validation.ValidationResult;

import java.util.ArrayList;
import java.util.List;
import java.util.Optional;

/**
 * Orientation-first layout rules for containment pen multiblocks.
 *
 * <p>Controller at the origin is the entrance on the bottom outer wall.
 * Discovery: outward → inward → left/right → depth → height → continuous enclosure.
 * {@link InteriorVolume} is the capacity SSOT.
 */
public final class ContainmentPenBlueprint {

    public static final int MIN_SIZE = 3;
    public static final int MAX_WIDTH = 9;
    public static final int MAX_DEPTH = 9;
    public static final int MAX_HEIGHT = 9;

    /** @deprecated Prefer extents from a validated {@link InteriorVolume}. */
    @Deprecated
    public static final double INTERIOR_EXTENT = ContainmentFitRules.DEFAULT_PEN_INTERIOR_EXTENT;

    private ContainmentPenBlueprint() {
    }

    public static ValidationResult validateLayout(int width, int depth) {
        return width >= MIN_SIZE && depth >= MIN_SIZE && width <= MAX_WIDTH && depth <= MAX_DEPTH
                ? ValidationResult.success()
                : ValidationResult.failure("invalid pen layout");
    }

    public static ContainmentPenValidationResult validate(OccupancyCheck check) {
        Optional<HorizontalFacing> outwardOpt = resolveOutward(check);
        if (outwardOpt.isEmpty()) {
            return ContainmentPenValidationResult.invalid("ambiguous or missing entrance outward face");
        }
        HorizontalFacing outward = outwardOpt.get();
        HorizontalFacing inward = outward.opposite();
        HorizontalFacing left = outward.left();
        HorizontalFacing right = outward.right();

        int leftExtent = span(check, left);
        int rightExtent = span(check, right);
        int width = leftExtent + rightExtent + 1;
        if (width < MIN_SIZE || width > MAX_WIDTH) {
            return ContainmentPenValidationResult.invalid("width out of range: " + width);
        }

        int depth = measureDepth(check, inward, left, right, leftExtent, rightExtent);
        if (depth < MIN_SIZE || depth > MAX_DEPTH) {
            return ContainmentPenValidationResult.invalid("depth out of range: " + depth);
        }

        Footprint foot = footprint(inward, left, right, leftExtent, rightExtent, depth);
        int height = measureHeight(check, foot);
        if (height < MIN_SIZE || height > MAX_HEIGHT) {
            return ContainmentPenValidationResult.invalid("height out of range: " + height);
        }

        if (!enclosureComplete(check, foot, height)) {
            return ContainmentPenValidationResult.invalid("enclosure incomplete");
        }

        InteriorVolume interior = new InteriorVolume(
                foot.minX + 1, 1, foot.minZ + 1,
                foot.maxX - 1, height - 2, foot.maxZ - 1);
        if (interior.sizeX() < 1 || interior.sizeY() < 1 || interior.sizeZ() < 1) {
            return ContainmentPenValidationResult.invalid("interior too small");
        }
        if (!interiorClear(check, interior)) {
            return ContainmentPenValidationResult.invalid("interior not empty");
        }

        List<ShellCell> shells = shellCells(foot, height);
        return new ContainmentPenValidationResult(
                true, List.copyOf(shells), interior, outward, width, depth, height, "ok");
    }

    private static Optional<HorizontalFacing> resolveOutward(OccupancyCheck check) {
        List<HorizontalFacing> open = new ArrayList<>(4);
        for (HorizontalFacing facing : HorizontalFacing.values()) {
            if (!check.isOccupied(facing.dx(), 0, facing.dz())) {
                open.add(facing);
            }
        }
        if (open.size() != 1) {
            return Optional.empty();
        }
        HorizontalFacing outward = open.get(0);
        HorizontalFacing inward = outward.opposite();
        boolean anchored = check.isOccupied(outward.left().dx(), 0, outward.left().dz())
                || check.isOccupied(outward.right().dx(), 0, outward.right().dz())
                || check.isOccupied(inward.dx(), 0, inward.dz());
        return anchored ? Optional.of(outward) : Optional.empty();
    }

    /** How far continuous floor shell extends from controller along {@code dir}. */
    private static int span(OccupancyCheck check, HorizontalFacing dir) {
        int extent = 0;
        for (int i = 1; i < MAX_WIDTH; i++) {
            if (!check.isOccupied(dir.dx() * i, 0, dir.dz() * i)) {
                break;
            }
            extent = i;
        }
        return extent;
    }

    private static int measureDepth(
            OccupancyCheck check,
            HorizontalFacing inward,
            HorizontalFacing left,
            HorizontalFacing right,
            int leftExtent,
            int rightExtent) {
        int depth = 1;
        for (int d = 1; d < MAX_DEPTH; d++) {
            if (!floorRowPresent(check, inward, left, right, leftExtent, rightExtent, d)) {
                break;
            }
            depth = d + 1;
        }
        return depth;
    }

    private static boolean floorRowPresent(
            OccupancyCheck check,
            HorizontalFacing inward,
            HorizontalFacing left,
            HorizontalFacing right,
            int leftExtent,
            int rightExtent,
            int depthIndex) {
        for (int l = 0; l <= leftExtent; l++) {
            int x = inward.dx() * depthIndex + left.dx() * l;
            int z = inward.dz() * depthIndex + left.dz() * l;
            if (depthIndex == 0 && l == 0) {
                continue;
            }
            if (!check.isOccupied(x, 0, z)) {
                return false;
            }
        }
        for (int r = 1; r <= rightExtent; r++) {
            int x = inward.dx() * depthIndex + right.dx() * r;
            int z = inward.dz() * depthIndex + right.dz() * r;
            if (!check.isOccupied(x, 0, z)) {
                return false;
            }
        }
        return true;
    }

    private static Footprint footprint(
            HorizontalFacing inward,
            HorizontalFacing left,
            HorizontalFacing right,
            int leftExtent,
            int rightExtent,
            int depth) {
        int minX = 0;
        int maxX = 0;
        int minZ = 0;
        int maxZ = 0;
        for (int d = 0; d < depth; d++) {
            for (int l = 0; l <= leftExtent; l++) {
                int x = inward.dx() * d + left.dx() * l;
                int z = inward.dz() * d + left.dz() * l;
                minX = Math.min(minX, x);
                maxX = Math.max(maxX, x);
                minZ = Math.min(minZ, z);
                maxZ = Math.max(maxZ, z);
            }
            for (int r = 1; r <= rightExtent; r++) {
                int x = inward.dx() * d + right.dx() * r;
                int z = inward.dz() * d + right.dz() * r;
                minX = Math.min(minX, x);
                maxX = Math.max(maxX, x);
                minZ = Math.min(minZ, z);
                maxZ = Math.max(maxZ, z);
            }
        }
        return new Footprint(minX, maxX, minZ, maxZ);
    }

    private static int measureHeight(OccupancyCheck check, Footprint foot) {
        int height = 1;
        for (int y = 1; y < MAX_HEIGHT; y++) {
            boolean perimeterOk = true;
            for (int x = foot.minX; x <= foot.maxX && perimeterOk; x++) {
                for (int z = foot.minZ; z <= foot.maxZ; z++) {
                    boolean onPerimeter = x == foot.minX || x == foot.maxX || z == foot.minZ || z == foot.maxZ;
                    if (onPerimeter && !check.isOccupied(x, y, z)) {
                        perimeterOk = false;
                        break;
                    }
                }
            }
            if (!perimeterOk) {
                break;
            }
            height = y + 1;
        }
        // Require a complete ceiling slab at top
        while (height >= MIN_SIZE) {
            int top = height - 1;
            boolean ceilingOk = true;
            for (int x = foot.minX; x <= foot.maxX && ceilingOk; x++) {
                for (int z = foot.minZ; z <= foot.maxZ; z++) {
                    if (!check.isOccupied(x, top, z)) {
                        ceilingOk = false;
                        break;
                    }
                }
            }
            if (ceilingOk) {
                return height;
            }
            height--;
        }
        return height;
    }

    private static boolean enclosureComplete(OccupancyCheck check, Footprint foot, int height) {
        int maxY = height - 1;
        for (int y = 0; y <= maxY; y++) {
            for (int x = foot.minX; x <= foot.maxX; x++) {
                for (int z = foot.minZ; z <= foot.maxZ; z++) {
                    if (x == 0 && y == 0 && z == 0) {
                        continue;
                    }
                    boolean shell = y == 0 || y == maxY
                            || x == foot.minX || x == foot.maxX
                            || z == foot.minZ || z == foot.maxZ;
                    if (shell && !check.isOccupied(x, y, z)) {
                        return false;
                    }
                }
            }
        }
        return true;
    }

    private static boolean interiorClear(OccupancyCheck check, InteriorVolume interior) {
        for (int y = interior.minY(); y <= interior.maxY(); y++) {
            for (int x = interior.minX(); x <= interior.maxX(); x++) {
                for (int z = interior.minZ(); z <= interior.maxZ(); z++) {
                    if (check.isOccupied(x, y, z)) {
                        return false;
                    }
                }
            }
        }
        return true;
    }

    private static List<ShellCell> shellCells(Footprint foot, int height) {
        int maxY = height - 1;
        List<ShellCell> cells = new ArrayList<>();
        for (int y = 0; y <= maxY; y++) {
            for (int x = foot.minX; x <= foot.maxX; x++) {
                for (int z = foot.minZ; z <= foot.maxZ; z++) {
                    if (x == 0 && y == 0 && z == 0) {
                        continue;
                    }
                    boolean shell = y == 0 || y == maxY
                            || x == foot.minX || x == foot.maxX
                            || z == foot.minZ || z == foot.maxZ;
                    if (!shell) {
                        continue;
                    }
                    ShellRole role = y == 0 ? ShellRole.FLOOR : (y == maxY ? ShellRole.CEILING : ShellRole.WALL);
                    cells.add(new ShellCell(x, y, z, role));
                }
            }
        }
        return cells;
    }

    private record Footprint(int minX, int maxX, int minZ, int maxZ) {
    }

    public record ContainmentPenValidationResult(
            boolean isValid,
            List<ShellCell> shellCells,
            InteriorVolume interior,
            HorizontalFacing outward,
            int width,
            int depth,
            int height,
            String reason) {

        public static ContainmentPenValidationResult invalid(String reason) {
            return new ContainmentPenValidationResult(false, List.of(), null, null, 0, 0, 0, reason);
        }

        @Deprecated
        public int structureHeight() {
            return height;
        }

        public List<BlockOffset> occupiedOffsets() {
            List<BlockOffset> list = new ArrayList<>(shellCells.size());
            for (ShellCell cell : shellCells) {
                list.add(new BlockOffset(cell.x(), cell.y(), cell.z(), true));
            }
            return list;
        }
    }

    public record BlockOffset(int x, int y, int z, boolean required) {
    }

    @FunctionalInterface
    public interface OccupancyCheck {
        boolean isOccupied(int offsetX, int offsetY, int offsetZ);
    }

    public static List<BlockPosRef> toWorldOffsets(BlockPosRef center, List<BlockOffset> offsets) {
        List<BlockPosRef> world = new ArrayList<>(offsets.size());
        for (BlockOffset offset : offsets) {
            world.add(center.offset(offset.x(), offset.y(), offset.z()));
        }
        return world;
    }

    public static List<BlockPosRef> shellCellsToWorld(BlockPosRef center, List<ShellCell> cells) {
        List<BlockPosRef> world = new ArrayList<>(cells.size());
        for (ShellCell cell : cells) {
            world.add(center.offset(cell.x(), cell.y(), cell.z()));
        }
        return world;
    }

/**
 * Ownership of the latched host for the attachment lifecycle.
 *
 * <p>Combat owns {@code Mob.getTarget()}. Attachment owns this binding. They are related but
 * must never be aliases — attachment systems must not use combat target to identify the host.
 */
public final class AttachmentBinding {

    @Nullable
    private LivingEntityRef host;

    public void bind(@Nullable LivingEntityRef host) {
        this.host = host;
    }

    @Nullable
    public LivingEntityRef host() {
        return host;
    }

    @Nullable
    public UUID hostUUID() {
        return host != null ? host.entityId() : null;
    }

    public boolean isBound() {
        return host != null;
    }

    public void clear() {
        this.host = null;
    }
}
